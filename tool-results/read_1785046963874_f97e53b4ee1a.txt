   315→// ─── Generating Phase (real API calls, real progress) ─────────────────────
   316→
   317→interface PageGenState {
   318→  id: 'home' | 'about' | 'services' | 'contact'
   319→  name: string
   320→  status: 'pending' | 'generating' | 'done' | 'error'
   321→  error?: string
   322→  html?: string
   323→  css?: string
   324→  js?: string
   325→  siteName?: string
   326→}
   327→
   328→function GeneratingPhase() {
   329→  const {
   330→    builderPrompt, builderIndustry, builderStyle,
   331→    setGeneratedPages, setGeneratedSiteName,
   332→    setCurrentPreviewPage, setIsGenerating, setBuilderPhase,
   333→    setGenerationProgress, setGenerationStatus,
   334→  } = useAppStore()
   335→
   336→  const [pageStates, setPageStates] = useState<PageGenState[]>(
   337→    PAGE_ORDER.map(p => ({ id: p.id, name: p.name, status: 'pending' }))
   338→  )
   339→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   340→  const [globalError, setGlobalError] = useState<string | null>(null)
   341→  const cancelRef = useRef(false)
   342→
   343→  // Tick elapsed timer
   344→  useEffect(() => {
   345→    const interval = setInterval(() => {
   346→      if (!cancelRef.current) setElapsedSeconds(s => s + 1)
   347→    }, 1000)
   348→    return () => clearInterval(interval)
   349→  }, [])
   350→
   351→  // Run the 4 page generations SEQUENTIALLY (parallel hits the GLM-4.7 API
   352→  // rate limit and returns 429). Each call takes ~60-90s, so total is ~5min.
   353→  // The user sees real per-page progress, so the wait feels productive.
   354→  useEffect(() => {
   355→    cancelRef.current = false
   356→    let cancelled = false
   357→
   358→    async function generateAll() {
   359→      const collected: { id: string; name: string; html: string; css: string; js?: string }[] = []
   360→      let siteName = ''
   361→
   362→      for (let i = 0; i < PAGE_ORDER.length; i++) {
   363→        if (cancelled || cancelRef.current) return
   364→        const pageInfo = PAGE_ORDER[i]
   365→
   366→        setPageStates(prev => prev.map(p => p.id === pageInfo.id ? { ...p, status: 'generating' } : p))
   367→        setGenerationStatus(`Generating ${pageInfo.name} page…`)
   368→        setGenerationProgress(Math.round((i / PAGE_ORDER.length) * 100))
   369→
   370→        try {
   371→          // POLLING APPROACH (fixes 502 from preview-proxy):
   372→          // 1. POST /api/generate → returns { jobId } immediately
   373→          // 2. Poll GET /api/generate?jobId=X every 2.5s until status is
   374→          //    'done' or 'error'
   375→          // 3. Each request is short (~100ms) so no proxy can 502 out
   376→          //
   377→          // The GLM-4.7 API takes 60-150s per page, but the work runs in
   378→          // the background on the server. The client just polls.
   379→          const startRes = await fetch('/api/generate', {
   380→            method: 'POST',
   381→            headers: { 'Content-Type': 'application/json' },
   382→            body: JSON.stringify({
   383→              prompt: builderPrompt,
   384→              industry: builderIndustry,
   385→              style: builderStyle,
   386→              page: pageInfo.id,
   387→            }),
   388→          })
   389→
   390→          if (cancelled || cancelRef.current) return
   391→
   392→          if (!startRes.ok) {
   393→            const errBody = await startRes.json().catch(() => ({ error: `HTTP ${startRes.status}` }))
   394→            throw new Error(errBody.error || `HTTP ${startRes.status}`)
   395→          }
   396→
   397→          const startData = await startRes.json()
   398→          const jobId = startData?.jobId
   399→          if (!jobId) {
   400→            throw new Error('Server did not return a jobId')
   401→          }
   402→
   403→          // Poll until done/error
   404→          let data: any = null
   405→          let streamError: string | null = null
   406→          const POLL_INTERVAL_MS = 2500
   407→          const MAX_POLL_DURATION_MS = 8 * 60 * 1000 // 8 min per page max
   408→          const pollStart = Date.now()
   409→
   410→          while (true) {
   411→            if (cancelled || cancelRef.current) return
   412→
   413→            const elapsed = Date.now() - pollStart
   414→            if (elapsed > MAX_POLL_DURATION_MS) {
   415→              throw new Error(`Timed out after ${Math.round(elapsed / 1000)}s waiting for ${pageInfo.name} page`)
   416→            }
   417→
   418→            // Wait before next poll (but not on first iteration)
   419→            if (elapsed > 0) {
   420→              await new Promise<void>(r => setTimeout(r, POLL_INTERVAL_MS))
   421→              if (cancelled || cancelRef.current) return
   422→            }
   423→
   424→            const statusRes = await fetch(`/api/generate?jobId=${encodeURIComponent(jobId)}`, {
   425→              method: 'GET',
   426→              headers: { 'Cache-Control': 'no-cache' },
   427→            })
   428→
   429→            if (!statusRes.ok) {
   430→              // 404 = job expired/gone
   431→              if (statusRes.status === 404) {
   432→                throw new Error('Generation job disappeared — please retry')
   433→              }
   434→              const errBody = await statusRes.json().catch(() => ({ error: `HTTP ${statusRes.status}` }))
   435→              throw new Error(errBody.error || `Status poll HTTP ${statusRes.status}`)
   436→            }
   437→
   438→            const status = await statusRes.json()
   439→            if (cancelled || cancelRef.current) return
   440→
   441→            // Update UI with heartbeat info so the user sees progress
   442→            if (status.heartbeats !== undefined) {
   443→              setGenerationStatus(
   444→                `Generating ${pageInfo.name} page… (${status.heartbeats} AI ticks, ${Math.round((Date.now() - pollStart) / 1000)}s)`
   445→              )
   446→            }
   447→
   448→            if (status.status === 'done') {
   449→              data = status
   450→              break
   451→            }
   452→            if (status.status === 'error') {
   453→              streamError = status.error || 'Generation failed'
   454→              break
   455→            }
   456→            // else: 'pending' or 'generating' — keep polling
   457→          }
   458→
   459→          if (cancelled || cancelRef.current) return
   460→
   461→          if (streamError) {
   462→            throw new Error(streamError)
   463→          }
   464→          if (!data?.result?.page?.html) {
   465→            throw new Error('AI returned no HTML')
   466→          }
   467→
   468→          if (data.result.siteName && !siteName) siteName = data.result.siteName
   469→
   470→          const completed: PageGenState = {
   471→            id: pageInfo.id,
   472→            name: data.result.page.name || pageInfo.name,
   473→            status: 'done',
   474→            html: data.result.page.html,
   475→            css: data.result.page.css || '',
   476→            js: data.result.page.js || '',
   477→          }
   478→
   479→          collected.push({
   480→            id: `page-${pageInfo.id}`,
   481→            name: completed.name,
   482→            route: pageInfo.id === 'home' ? '/' : `/${pageInfo.id}`,
   483→            html: completed.html || '',
   484→            css: completed.css || '',
   485→            js: completed.js,
   486→          })
   487→
   488→          setPageStates(prev => prev.map(p => p.id === pageInfo.id ? completed : p))
   489→          setGenerationProgress(Math.round(((i + 1) / PAGE_ORDER.length) * 100))
   490→        } catch (err: any) {
   491→          if (cancelled || cancelRef.current) return
   492→          const msg = err?.message || 'Generation failed'
   493→          setPageStates(prev => prev.map(p => p.id === pageInfo.id ? { ...p, status: 'error', error: msg } : p))
   494→          setGlobalError(msg)
   495→          return
   496→        }
   497→      }
   498→
   499→      if (cancelled || cancelRef.current) return
   500→
   501→      setGeneratedPages(collected)
   502→      setGeneratedSiteName(siteName || 'Untitled Site')
   503→      setCurrentPreviewPage(collected[0].id)
   504→      setGenerationProgress(100)
   505→      setGenerationStatus('Done!')
   506→      setIsGenerating(false)
   507→      setBuilderPhase('preview')
   508→    }
   509→
   510→    generateAll()
   511→
   512→    return () => {
   513→      cancelled = true
   514→      cancelRef.current = true
   515→    }
   516→    // eslint-disable-next-line react-hooks/exhaustive-deps
   517→  }, [])
   518→
   519→  const handleCancel = () => {
   520→    cancelRef.current = true
   521→    setIsGenerating(false)
   522→    setGenerationProgress(0)
   523→    setGenerationStatus('')
   524→    setBuilderPhase('prompt')
   525→  }
   526→
   527→  const handleRetry = () => {
   528→    // Reset states and re-run by remounting (force re-render via phase change)
   529→    setIsGenerating(false)
   530→    setGenerationProgress(0)
   531→    setGenerationStatus('')
   532→    setBuilderPhase('prompt')
   533→  }
   534→
   535→  const formatTime = (s: number) => `${Math.floor(s / 60)}:${(s % 60).toString().padStart(2, '0')}`
   536→  const doneCount = pageStates.filter(p => p.status === 'done').length
   537→  const hasError = pageStates.some(p => p.status === 'error')
   538→
   539→  return (

   691→            variant="ghost"
   692→            onClick={handleCancel}

   704→          </p>

   707→    </div>
