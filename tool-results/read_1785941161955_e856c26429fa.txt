    86→/** Find a DOM node by its path-based fid (e.g. "0-1-2") inside root. */
    87→export function findByFid(root: HTMLElement | null, fid: string): HTMLElement | null {
    88→  if (!root) return null
    89→  const parts = fid.split('-').map(Number)
    90→  let el: Element = root
    91→  for (const idx of parts) {
    92→    const children = Array.from(el.children).filter((c) => c.nodeType === 1)
    93→    if (idx >= children.length) return null
    94→    el = children[idx]
    95→  }
    96→  return el as HTMLElement
    97→}

    99→// ═══════════════════════════════════════════════════════════════════════════
   100→// buildFid — stable path id from element to root
   101→// ═══════════════════════════════════════════════════════════════════════════
   102→
   103→function buildFid(el: HTMLElement, root: HTMLElement): string {
   104→  const parts: string[] = []
   105→  let node: HTMLElement | null = el
   106→  while (node && node !== root) {
   107→    const parent = node.parentElement
   108→    let index = 0
   109→    if (parent) {
   110→      for (let sib: ChildNode | null = node.previousSibling; sib; sib = sib.previousSibling) {
   111→        if (sib.nodeType === 1) index++
   112→      }
   113→    }
   114→    parts.unshift(String(index))
   115→    node = parent
   116→  }
   117→  return parts.join('-')
   118→}

   188→// ═══════════════════════════════════════════════════════════════════════════
   189→// Resize Handles Overlay (React component for more control)
   190→// ═══════════════════════════════════════════════════════════════════════════
   191→
   192→interface HandlePosition {
   193→  key: string
   194→  style: React.CSSProperties
   195→  cursor: string
   196→}
   197→
   198→function useResizeHandles(rect: DOMRect | null, zoom: number): HandlePosition[] {
   199→  return useMemo(() => {
   200→    if (!rect || rect.width === 0 || rect.height === 0) return []
   201→
   202→    const HANDLE_SIZE = 8
   203→    const half = HANDLE_SIZE / 2
   204→    // Handles are positioned relative to the selection rect
   205→    const top = 0
   206→    const left = 0
   207→    const right = rect.width
   208→    const bottom = rect.height
   209→    const midX = rect.width / 2
   210→    const midY = rect.height / 2
   211→
   212→    return [
   213→      { key: 'nw', cursor: 'nwse-resize', style: { top: top - half, left: left - half } },
   214→      { key: 'n', cursor: 'ns-resize', style: { top: top - half, left: midX - half } },
   215→      { key: 'ne', cursor: 'nesw-resize', style: { top: top - half, left: right - half } },
   216→      { key: 'w', cursor: 'ew-resize', style: { top: midY - half, left: left - half } },
   217→      { key: 'e', cursor: 'ew-resize', style: { top: midY - half, left: right - half } },
   218→      { key: 'sw', cursor: 'nesw-resize', style: { top: bottom - half, left: left - half } },
   219→      { key: 's', cursor: 'ns-resize', style: { top: bottom - half, left: midX - half } },
   220→      { key: 'se', cursor: 'nwse-resize', style: { top: bottom - half, left: right - half } },
   221→    ].map((h) => ({
   222→      ...h,
   223→      style: {
   224→        ...h.style,
   225→        position: 'absolute' as const,
   226→        width: HANDLE_SIZE,
   227→        height: HANDLE_SIZE,
   228→        background: '#FFFFFF',
   229→        border: '1.5px solid #3B82F6',
   230→        borderRadius: 1,
   231→        zIndex: Z_INDEX.selection + 2,
   232→        pointerEvents: 'auto' as const,
   233→        cursor: h.cursor,
   234→        // Subtle shadow for visibility on any background
   235→        boxShadow: '0 1px 3px rgba(0,0,0,0.15)',
   236→      },
   237→    }))
   238→  }, [rect, zoom])
   239→}

   242→// Canvas Component
   243→// ═══════════════════════════════════════════════════════════════════════════
   244→
   245→export function Canvas(p: CanvasProps) {
   246→  const theme = useEditorTheme()
   247→  const { reduceMotion } = useAccessibility()
   248→  const isDark = theme.mode === 'dark'
   249→
   250→  // ── Refs ──────────────────────────────────────────────────────────────
   251→  const vpRef = useRef<HTMLDivElement>(null)
   252→  const contentRef = useRef<HTMLDivElement>(null)
   253→  const panning = useRef(false)
   254→  const [isPanning, setIsPanning] = useState(false)
   255→  const lastPoint = useRef({ x: 0, y: 0 })
   256→  const selToken = useRef<string | null>(null)
   257→  const [selRect, setSelRect] = useState<DOMRect | null>(null)
   258→  const [contentRect, setContentRect] = useState<DOMRect | null>(null)
   259→  const [hoveredFid, setHoveredFid] = useState<string | null>(null)
   260→  const hoveredFidRef = useRef<string | null>(null)
   261→  const [showGrid, setShowGrid] = useState(true)
   262→
   263→  const width = DEVICE_WIDTH[p.device]
   264→  const bezel = DEVICE_BEZEL[p.device]
   265→
   266→  // ── Theme-derived colors ──────────────────────────────────────────────
   267→  const colors = isDark ? DARK_COLORS : LIGHT_COLORS
   268→  const canvasBg = colors.canvas.background
   269→  const canvasDot = colors.canvas.dot
   270→  const canvasFrame = colors.canvas.frame
   271→  const artboardBg = colors.canvas.artboard
   272→  const selectionColor = colors.selection.fill
   273→  const handleFill = colors.selection.handle
   274→  const borderColor = colors.border.default
   275→  const textTertiary = colors.onSurface.tertiary
   276→
   277→  // ── Editor CSS (theme-aware) ──────────────────────────────────────────
   278→  const editorCss = useMemo(
   279→    () => buildEditorCss(selectionColor, handleFill),
   280→    [selectionColor, handleFill]
   281→  )
   282→  const fullCss = editorCss + (p.css || '')
   283→
   284→  // ── Dot grid background ──────────────────────────────────────────────
   285→  const gridStyle = useMemo(
   286→    () => ({
   287→      backgroundImage: `
   288→        radial-gradient(circle ${GRID.dotSize}px at center, ${canvasDot} 0.8px, transparent 0.8px),
   289→        radial-gradient(circle ${GRID.dotSize * 1.5}px at center, ${canvasDot} 0.4px, transparent 0.4px)
   290→      `,
   291→      backgroundSize: `${GRID.minorGrid}px ${GRID.minorGrid}px, ${GRID.majorGrid}px ${GRID.majorGrid}px`,
   292→    }),
   293→    [canvasDot]
   294→  )
   295→
   296→  // ── Resize handles ───────────────────────────────────────────────────
   297→  const handles = useResizeHandles(selRect, p.zoom)
   298→
   299→  // ── Selection highlight sync ─────────────────────────────────────────
   300→  useEffect(() => {
   301→    const root = contentRef.current
   302→    if (!root) return
   303→
   304→    // Clear previous
   305→    if (selToken.current) {
   306→      const prev = root.querySelector(`[data-ve-sel="${selToken.current}"]`)
   307→      if (prev) {
   308→        prev.classList.remove('ve-selected')
   309→        prev.removeAttribute('data-ve-sel')
   310→      }
   311→      selToken.current = null
   312→      setSelRect(null)
   313→      setContentRect(null)
   314→    }
   315→
   316→    if (!p.selectedFid) return
   317→
   318→    const el = findByFid(root, p.selectedFid)
   319→    if (el) {
   320→      el.classList.add('ve-selected')
   321→      el.setAttribute('data-ve-sel', p.selectedFid)
   322→      selToken.current = p.selectedFid
   323→
   324→      const rect = el.getBoundingClientRect()
   325→      setSelRect(rect)
   326→      const cRect = contentRef.current?.getBoundingClientRect() || null
   327→      setContentRect(cRect)
   328→
   329→      // Scroll to selected element if off-screen
   330→      const vp = vpRef.current
   331→      if (vp) {
   332→        const vpRect = vp.getBoundingClientRect()
   333→        const margin = 80
   334→        if (
   335→          rect.top < vpRect.top + margin ||
   336→          rect.bottom > vpRect.bottom - margin ||
   337→          rect.left < vpRect.left + margin ||
   338→          rect.right > vpRect.right - margin
   339→        ) {
   340→          el.scrollIntoView({
   341→            behavior: reduceMotion ? 'auto' : 'smooth',
   342→            block: 'nearest',
   343→            inline: 'nearest',
   344→          })
   345→        }
   346→      }
   347→    }
   348→  }, [p.selectedFid, p.html, reduceMotion])
   349→
   350→  // ── Expose content root ──────────────────────────────────────────────
   351→  useEffect(() => {
   352→    p.onContentReady?.(contentRef.current)
   353→  }, [p.onContentReady])
   354→
   355→  // ── Attach hover outlines to selectable elements ─────────────────────
   356→  useEffect(() => {
   357→    const root = contentRef.current
   358→    if (!root) return
   359→    const elements = root.querySelectorAll(SELECTABLE)
   360→    elements.forEach((el) => {
   361→      ;(el as HTMLElement).classList.add('ve-hover-target')
   362→    })
   363→    return () => {
   364→      elements.forEach((el) => {
   365→        ;(el as HTMLElement).classList.remove('ve-hover-target')
   366→      })
   367→    }
   368→  }, [p.html])
   369→
   370→  // ── Track hover for dashed indicator ─────────────────────────────────
   371→  const handleMouseMove = useCallback(
   372→    (e: React.MouseEvent) => {
   373→      if (panning.current) return
   374→      const root = contentRef.current
   375→      if (!root) return
   376→
   377→      const target = e.target as HTMLElement
   378→      const el = target.closest(SELECTABLE) as HTMLElement | null
   379→
   380→      // Clear previous hover
   381→      if (hoveredFidRef.current) {
   382→        const prev = findByFid(root, hoveredFidRef.current)
   383→        if (prev && prev !== el) {
   384→          prev.classList.remove('ve-hovered')
   385→        }
   386→      }
   387→
   388→      if (el && root.contains(el) && el.classList.contains('ve-hover-target')) {
   389→        const fid = buildFid(el, root)
   390→        if (fid !== p.selectedFid) {
   391→          el.classList.add('ve-hovered')
   392→          hoveredFidRef.current = fid
   393→          setHoveredFid(fid)
   394→        } else {
   395→          hoveredFidRef.current = null
   396→          setHoveredFid(null)
   397→        }
   398→      } else {
   399→        hoveredFidRef.current = null
   400→        setHoveredFid(null)
   401→      }
   402→    },
   403→    [p.selectedFid]
   404→  )
   405→
   406→  const handleMouseLeave = useCallback(() => {
   407→    const root = contentRef.current
   408→    if (root && hoveredFidRef.current) {
   409→      const prev = findByFid(root, hoveredFidRef.current)
   410→      if (prev) prev.classList.remove('ve-hovered')
   411→    }
   412→    hoveredFidRef.current = null
   413→    setHoveredFid(null)
   414→  }, [])
   415→
   416→  // ── Read element info ────────────────────────────────────────────────
   417→  const readInfo = useCallback((el: HTMLElement, root: HTMLElement): SelectionInfo | null => {
   418→    const fid = buildFid(el, root)
   419→    const tag = el.tagName.toLowerCase()
   420→    const cs = window.getComputedStyle(el)
   421→    const rect = el.getBoundingClientRect()
   422→    const isSection = ['section', 'header', 'footer', 'nav', 'ul', 'li', 'form', 'div'].includes(tag)
   423→    return {
   424→      fid,
   425→      tag,
   426→      text: (el.textContent || '').trim().slice(0, 120),
   427→      isImage: tag === 'img',
   428→      isText: ['h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'p', 'a', 'span', 'li'].includes(tag),
   429→      isButton: tag === 'button',
   430→      isLink: tag === 'a',
   431→      isSection,
   432→      color: cs.color,
   433→      bgColor:
   434→        cs.backgroundColor && cs.backgroundColor !== 'rgba(0, 0, 0, 0)'
   435→          ? cs.backgroundColor
   436→          : undefined,
   437→      fontSize: cs.fontSize,
   438→      fontWeight: cs.fontWeight,
   439→      rect: { left: rect.left, top: rect.top, width: rect.width, height: rect.height },
   440→    }
   441→  }, [])
   442→
   443→  // ── Click to select ──────────────────────────────────────────────────
   444→  const handleClick = useCallback(
   445→    (e: React.MouseEvent) => {
   446→      if (panning.current) return
   447→      const root = contentRef.current
   448→      if (!root) return
   449→      const target = e.target as HTMLElement
   450→      const el = target.closest(SELECTABLE) as HTMLElement | null
   451→      if (el && root.contains(el)) {
   452→        p.onSelect(readInfo(el, root))
   453→      } else {
   454→        p.onSelect(null)
   455→      }
   456→    },
   457→    [p.onSelect, readInfo]
   458→  )
   459→
   460→  // ── Drag-to-pan (Space + drag or middle mouse) ──────────────────────
   461→  const startPan = useCallback(
   462→    (e: React.PointerEvent) => {
   463→      if (e.button !== 1 && !p.spaceHeld) return
   464→      e.preventDefault()
   465→      e.stopPropagation()
   466→      panning.current = true
   467→      setIsPanning(true)
   468→      lastPoint.current = { x: e.clientX, y: e.clientY }
   469→
   470→      const vp = vpRef.current
   471→
   472→      const move = (ev: PointerEvent) => {
   473→        if (!vp) return
   474→        const dx = ev.clientX - lastPoint.current.x
   475→        const dy = ev.clientY - lastPoint.current.y
   476→        vp.scrollLeft -= dx
   477→        vp.scrollTop -= dy
   478→        lastPoint.current = { x: ev.clientX, y: ev.clientY }
   479→      }
   480→      const up = () => {
   481→        panning.current = false
   482→        setIsPanning(false)
   483→        window.removeEventListener('pointermove', move)
   484→        window.removeEventListener('pointerup', up)
   485→      }
   486→      window.addEventListener('pointermove', move)
   487→      window.addEventListener('pointerup', up)
   488→    },
   489→    [p.spaceHeld]
   490→  )
   491→
   492→  // ── Keyboard: nudge + delete ─────────────────────────────────────────
   493→  useEffect(() => {
   494→    if (!p.selectedFid) return
   495→    const root = contentRef.current
   496→    if (!root) return
   497→    const el = findByFid(root, p.selectedFid)
   498→    if (!el) return
   499→
   500→    const onKey = (e: KeyboardEvent) => {
   501→      if (e.key.startsWith('Arrow')) {
   502→        e.preventDefault()
   503→        const dx = e.shiftKey ? 10 : 1
   504→        const m = /translate\(([-\d.]+)px, ([-\d.]+)px\)/.exec(el.style.transform || '')
   505→        const tx = m ? parseFloat(m[1]) : 0
   506→        const ty = m ? parseFloat(m[2]) : 0
   507→        let nx = tx,
   508→          ny = ty
   509→        if (e.key === 'ArrowLeft') nx -= dx
   510→        else if (e.key === 'ArrowRight') nx += dx
   511→        else if (e.key === 'ArrowUp') ny -= dx
   512→        else if (e.key === 'ArrowDown') ny += dx
   513→        el.style.transform = nx === 0 && ny === 0 ? '' : `translate(${nx}px, ${ny}px)`
   514→        announce(`${p.selectedFid} nudge ${dx}px`)
   515→      } else if (e.key === 'Delete' || e.key === 'Backspace') {
   516→        if (el.parentElement && el.parentElement !== root) {
   517→          const tag = el.tagName.toLowerCase()
   518→          el.remove()
   519→          announce(`Deleted ${tag}`)
   520→          p.onSelect(null)
   521→          setTimeout(() => p.onChangeHtml(root.innerHTML, `Delete ${tag}`), 0)
   522→        } else {
   523→          e.preventDefault()
   524→        }
   525→      }
   526→    }
   527→    window.addEventListener('keydown', onKey)
   528→    return () => window.removeEventListener('keydown', onKey)
   529→  }, [p.selectedFid, p.onChangeHtml, p.onSelect])
   530→
   531→  // ── Grid toggle keyboard shortcut (Ctrl+G) ───────────────────────────
   532→  useEffect(() => {
   533→    const onKey = (e: KeyboardEvent) => {
   534→      if ((e.ctrlKey || e.metaKey) && e.key === 'g') {
   535→        e.preventDefault()
   536→        setShowGrid((prev) => !prev)
   537→        announce(showGrid ? 'Grid hidden' : 'Grid visible')
   538→      }
   539→    }
   540→    window.addEventListener('keydown', onKey)
   541→    return () => window.removeEventListener('keydown', onKey)
   542→  }, [showGrid])
   543→
   544→  // ── Device frame label ───────────────────────────────────────────────
   545→  const deviceLabel = useMemo(() => {

   552→  const vpCursor = isPanning ? 'grabbing' : p.spaceHeld ? 'grab' : 'default'
   553→

   571→        overflow: 'auto',
   572→        background: canvasBg,

   578→    >

   581→        style={{
   582→          minWidth: '100%',
   583→          minHeight: '100%',
   584→          position: 'relative',
   585→          padding: '64px 48px',
   586→          display: 'flex',
   587→          flexDirection: 'column',
   588→          alignItems: 'center',
   589→        }}

   591→        {/* Device frame label */}
   592→        <div
   593→          className="flex items-center justify-center gap-2 mb-3 select-none"
   594→          style={{ zIndex: Z_INDEX.hover }}
   595→        >
   596→          <span
   597→            style={{
   598→              fontSize: 11,
   599→              fontWeight: 500,
   600→              fontFamily:
   601→                'Inter, ui-sans-serif, system-ui, -apple-system, sans-serif',
   602→              color: textTertiary,
   603→              background: isDark ? 'rgba(255,255,255,0.06)' : 'rgba(0,0,0,0.04)',
   604→              padding: '3px 10px',
   605→              borderRadius: RADIUS.md,
   606→              border: `1px solid ${borderColor}`,
   607→              letterSpacing: '0.02em',
   608→            }}
   609→          >
   610→            {deviceLabel} &middot; {width}px
   611→          </span>
   612→

   614→          <button
   615→            onClick={(e) => {
   616→              e.stopPropagation()
   617→              setShowGrid((prev) => !prev)
   618→            }}
   619→            title={showGrid ? 'Hide grid (Ctrl+G)' : 'Show grid (Ctrl+G)'}
   620→            aria-label={showGrid ? 'Hide grid' : 'Show grid'}
   621→            style={{
   622→              fontSize: 10,
   623→              fontWeight: 500,
   624→              fontFamily:
   625→                'Inter, ui-sans-serif, system-ui, -apple-system, sans-serif',
   626→              color: showGrid
   627→                ? selectionColor
   628→                : textTertiary,
   629→              background: showGrid
   630→                ? (isDark ? 'rgba(96,165,250,0.12)' : 'rgba(59,130,246,0.08)')
   631→                : (isDark ? 'rgba(255,255,255,0.06)' : 'rgba(0,0,0,0.04)'),
   632→              padding: '3px 8px',
   633→              borderRadius: RADIUS.md,
   634→              border: `1px solid ${showGrid ? selectionColor : borderColor}`,
   635→              cursor: 'pointer',
   636→              transition: reduceMotion ? 'none' : ANIMATION.transition.colors,
   637→            }}
   638→          >
   639→            Grid
   640→          </button>
   641→        </div>
   642→
   643→        {/* Device frame wrapper */}
   644→        <div
   645→          style={{
   646→            position: 'relative',
   647→            width: width + bezel * 2,
   648→            borderRadius: DEVICE_FRAME_RADIUS[p.device],
   649→            overflow: 'hidden',
   650→            // Mobile/tablet device bezel
   651→            border:
   652→              bezel > 0

   733→        </div>

   738→        className="select-none"

   750→          fontSize: 11,

   756→          borderRadius: RADIUS.md,
   757→          border: `1px solid ${borderColor}`,
   758→          zIndex: Z_INDEX.tooltip,

   760→          letterSpacing: '0.01em',

   762→      >
   763→        <svg

   771→          <line x1="6" y1="1" x2="6" y2="3.5" stroke="currentColor" strokeWidth="1.2" />
   772→          <line x1="6" y1="8.5" x2="6" y2="11" stroke="currentColor" strokeWidth="1.2" />
   773→          <line x1="1" y1="6" x2="3.5" y2="6" stroke="currentColor" strokeWidth="1.2" />
