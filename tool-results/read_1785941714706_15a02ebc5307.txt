   243→function LayersPanel({ htmlContent, onInsert, darkMode }: {
   244→  htmlContent?: string
   245→  onInsert: (html: string, label: string) => void
   246→  darkMode?: boolean
   247→}) {
   248→  const c = darkMode ? DARK_COLORS : COLORS
   249→
   250→  // Parse top-level elements from HTML
   251→  const elements = React.useMemo(() => {
   252→    if (!htmlContent) return []
   253→    try {
   254→      const parser = new DOMParser()
   255→      const doc = parser.parseFromString(htmlContent, 'text/html')
   256→      return Array.from(doc.body.children).map((el, i) => ({
   257→        tag: el.tagName.toLowerCase(),
   258→        text: (el.textContent || '').trim().slice(0, 40),
   259→        index: i,
   260→      }))
   261→    } catch {
   262→      return []
   263→    }
   264→  }, [htmlContent])
   265→
   266→  if (elements.length === 0) {
   267→    return (
