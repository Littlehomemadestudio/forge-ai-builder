  1121→function GeneratingPhase() {

  1133→    builderAdvancedOptions.pageConfigs.find(pc => pc.id === cp.id && pc.enabled)

  1137→    pagesToGenerate.map(p => ({ id: p.id, name: p.name, status: 'pending' }))

  1145→    const interval = setInterval(() => {
  1146→      if (!cancelRef.current) setElapsedSeconds(s => s + 1)
  1147→    }, 1000)
  1148→    return () => clearInterval(interval)

  1150→

  1175→            siteName: builderAdvancedOptions.brandName || undefined,

  1315→          setGlobalError(msg)

  1317→        }

  1319→
  1320→      if (cancelled || cancelRef.current) return
