    32→/** Find a DOM node by its escaped path fid inside the canvas root. */
    33→function findByFid(root: HTMLElement | null, fid: string): HTMLElement | null {
    34→  if (!root) return null
    35→  const parts = fid.split('-').map(Number)
    36→  let el: Element = root
    37→  for (const idx of parts) {
    38→    const children = Array.from(el.children).filter((c) => c.nodeType === 1)
    39→    if (idx >= children.length) return null
    40→    el = children[idx]
    41→  }
    42→  return el as HTMLElement
    43→}

   206→        method: 'POST',

   238→        while ((delim = buffer.match(/\r?\n\r?\n/)) !== null) {
   239→          const rawEvent = buffer.slice(0, delim.index!)
   240→          buffer = buffer.slice(delim.index! + delim[0].length)
   241→
   242→          let ev = 'message'
   243→          const dataLines: string[] = []
   244→          for (const line of rawEvent.split(/\r?\n/)) {
   245→            if (line.startsWith('event:')) ev = line.slice(6).trim()
   246→            else if (line.startsWith('data:')) dataLines.push(line.slice(5).trim())
   247→          }
   248→          if (dataLines.length === 0) continue
   249→          const dataStr = dataLines.join('\n')
   250→          let data: unknown
   251→          try { data = JSON.parse(dataStr) } catch { data = dataStr }
   252→
   253→          switch (ev) {
   254→            case 'delta': {
   255→              const chunk = (data as { chunk?: string })?.chunk || ''
   256→              streamingText += chunk
   257→              setAiFeedback(streamingText.length > 200 ? streamingText.slice(0, 200) + '…' : streamingText || 'AI is thinking…')
   258→              break
   259→            }
   260→            case 'result':
   261→              suggestion = data as typeof suggestion
   262→              break consume
   263→            case 'error':
   264→              {
   265→                const errData = data as { message?: string; error?: string }
   266→                const m = errData?.message || errData?.error || 'AI failed.'
   267→                setAiFeedback(m)
   268→                announce(m)
   269→                return
   270→              }
   271→            case 'done':
   272→              break consume
   273→            default:
   274→              break
   275→          }
   276→        }
   277→      }
   278→
   279→      // Apply AI-generated HTML if present
   280→      if (suggestion?.suggestions?.[0]?.newHtml) {
   281→        const newHtml = suggestion.suggestions[0].newHtml
   282→        if (node && node.parentElement && node.parentElement !== contentRootRef.current) {
   283→          // Replace selected element
   284→          const temp = document.createElement('div')

   296→            root.innerHTML += newHtml
   297→            commit(root.innerHTML, 'AI: append content')
   298→          }
   299→        }
   300→      }

   449→  }, [selection])
   450→
   451→  return (
   452→    <div className={`ve-shell${darkMode ? ' ve-dark' : ''}`} style={{ height: '100vh', width: '100vw', display: 'flex', flexDirection: 'column', overflow: 'hidden', fontSize: baseFont }}>
   453→      <LiveRegion />
   454→
   455→      {/* Saving indicator */}
   456→      {saving && (
   457→        <div aria-live="polite" style={{ position: 'fixed', top: 60, right: 16, zIndex: 600, background: COLORS.primary, color: '#FFF', padding: '6px 14px', borderRadius: RADIUS.lg, fontSize: 12, fontWeight: 500, display: 'flex', alignItems: 'center', gap: 6 }}>
   458→          <span className="ve-spinner" style={{ width: 12, height: 12, border: '2px solid rgba(255,255,255,0.3)', borderTopColor: '#FFF', borderRadius: '50%', display: 'inline-block' }} />
   459→          Saving…
   460→        </div>
   461→      )}
   462→
   463→      <TopNav
   464→        projectName={projectName}
   465→        onProjectNameChange={setProjectName}
   466→        canUndo={historyIndex > 0}
   467→        canRedo={historyIndex < history.length - 1}
   468→        onUndo={undo}
   469→        onRedo={redo}
   470→        onBack={() => navigate('dashboard')}
   471→        device={device}
   472→        onDeviceChange={setDevice}
   473→        zoom={zoom}
   474→        onZoomIn={() => setZoom((z) => Math.min(200, z + 10))}
   475→        onZoomOut={() => setZoom((z) => Math.max(20, z - 10))}
   476→        onFit={() => setZoom(70)}
   477→        onCommandPalette={() => setCmdOpen(true)}
   478→        onShortcuts={() => setShortcutsOpen(true)}
   479→        onPublish={onPublish}
   480→        onToggleToolbar={() => setToolbarVisible((v) => !v)}
   481→      />
   482→
   483→      <div style={{ flex: 1, display: 'flex', minHeight: 0 }}>
   484→        {toolbarVisible && (
   485→          <IconToolbar
   486→            active={activeTool}
   487→            onSelect={(id) => {
   488→              setActiveTool(id)
   489→              if (id === 'ai') announce('AI: describe a request in the assistant bar below')
   490→            }}
   491→          />
   492→        )}
   493→
   494→        <div style={{ flex: 1, display: 'flex', flexDirection: 'column', minWidth: 0 }}>
   495→          <div style={{ flex: 1, display: 'flex', position: 'relative', minHeight: 0 }}>
   496→            {hasContent ? (
   497→              <Canvas
   498→                html={html}
   499→                css={css}
   500→                device={device}
   501→                zoom={zoom}
   502→                selectedFid={selection?.fid || null}
   503→                spaceHeld={spaceHeld}
   504→                onSelect={setSelection}
   505→                onChangeHtml={commit}
   506→                hasContent={hasContent}
   507→                onContentReady={(el) => { contentRootRef.current = el }}
   508→              />
   509→            ) : (
   510→              <div style={{ flex: 1, background: darkMode ? DARK_COLORS.background : COLORS.background, overflow: 'auto' }}>
   511→                <EmptyCanvas onAction={onEmptyAction} />
   512→              </div>
   513→            )}
   514→
   515→            <FloatingSelectionBar
   516→              selection={selection}
   517→              onDuplicate={duplicate}
   518→              onDelete={remove}
   519→              onAI={() => runAI('Improve this element')}
   520→              onBold={bold}
   521→              onAlign={align}
   522→            />
   523→          </div>
   524→
   525→          <AIAssistantBar
   526→            onSubmit={(prompt) => runAI(prompt)}
   527→            isBusy={aiBusy}
   528→            onStop={stopAI}
   529→            onRegenerate={() => runAI('Improve this element')}
   530→            lastResponse={aiFeedback}
   531→            suggestions={aiSuggestions}
   532→          />
   533→          {aiFeedback && (
   534→            <div role="status" aria-live="polite" style={{ padding: '8px 16px', background: darkMode ? DARK_COLORS.primaryLight : COLORS.primaryLight, color: darkMode ? DARK_COLORS.primary : COLORS.primary, fontSize: 13, borderTop: `1px solid ${darkMode ? DARK_COLORS.selectionLight : COLORS.selectionLight}` }}>
   535→              ✨ {aiFeedback}
   536→            </div>
   537→          )}
   538→          {auditResults.length > 0 && (
   539→            <div role="status" style={{ padding: '8px 16px', background: darkMode ? DARK_COLORS.warningLight : COLORS.warningLight, color: '#92400E', fontSize: 13, borderTop: `1px solid ${darkMode ? DARK_COLORS.warning : COLORS.warning}` }}>
   540→              <strong style={{ fontWeight: 600 }}>A11y audit:</strong> {auditResults.map((r) => <span key={r} style={{ display: 'inline-block', marginRight: 12 }}>• {r}</span>)}
   541→            </div>
   542→          )}
   543→        </div>
   544→
   545→        {inspectorOpen && (
   546→          <Inspector
   547→            selection={selection}
   548→            onApply={commitNode}
   549→            onSelectText={onSelectText}
   550→            onDeleteSelection={remove}
   551→            onDuplicateSelection={duplicate}
   552→            onRunAccessibilityAudit={runAudit}
   553→            onFixAccessibility={fixAccessibility}
   554→            onExport={onExport}
   555→            onAI={(prompt) => runAI(prompt)}
   556→            darkMode={darkMode}
   557→            onToggleDarkMode={() => setDarkMode((d) => !d)}
   558→            onToggleInspector={() => setInspectorOpen(false)}
   559→          />
   560→        )}
   561→      </div>
   562→
   563→      <CommandPalette open={cmdOpen} onClose={() => setCmdOpen(false)} onRun={runCommand} />
   564→      <ShortcutsHelp open={shortcutsOpen} onClose={() => setShortcutsOpen(false)} />
   565→
   566→      {toast && (
   567→        <div role="status" style={{ position: 'fixed', bottom: 80, left: '50%', transform: 'translateX(-50%)', background: darkMode ? DARK_COLORS.text : COLORS.text, color: '#FFF', padding: '10px 18px', borderRadius: RADIUS.lg, boxShadow: SHADOWS.lg, zIndex: 500, fontSize: 14, fontWeight: 500 }}>
   568→          {toast}
   569→        </div>
   570→      )}
   571→    </div>
   572→  )
   573→}
