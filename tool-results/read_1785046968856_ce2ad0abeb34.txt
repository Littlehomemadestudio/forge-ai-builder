   328→   328→function GeneratingPhase() {
   329→   329→  const {
   330→   330→    builderPrompt, builderIndustry, builderStyle,
   331→   331→    setGeneratedPages, setGeneratedSiteName,
   332→   332→    setCurrentPreviewPage, setIsGenerating, setBuilderPhase,
   333→   333→    setGenerationProgress, setGenerationStatus,
   334→   334→  } = useAppStore()
   335→   335→
   336→   336→  const [pageStates, setPageStates] = useState<PageGenState[]>(
   337→   337→    PAGE_ORDER.map(p => ({ id: p.id, name: p.name, status: 'pending' }))
   338→   338→  )
   339→   339→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
   340→   340→  const [globalError, setGlobalError] = useState<string | null>(null)
   341→   341→  const cancelRef = useRef(false)
   342→   342→
   343→   343→  // Tick elapsed timer
   344→   344→  useEffect(() => {
   345→   345→    const interval = setInterval(() => {
   346→   346→      if (!cancelRef.current) setElapsedSeconds(s => s + 1)
   347→   347→    }, 1000)
   348→   348→    return () => clearInterval(interval)
   349→   349→  }, [])
   350→   350→
   351→   351→  // Run the 4 page generations SEQUENTIALLY (parallel hits the GLM-4.7 API
   352→   352→  // rate limit and returns 429). Each call takes ~60-90s, so total is ~5min.
   353→   353→  // The user sees real per-page progress, so the wait feels productive.
   354→   354→  useEffect(() => {
   355→   355→    cancelRef.current = false
   356→   356→    let cancelled = false
   357→   357→
   358→   358→    async function generateAll() {
   359→   359→      const collected: { id: string; name: string; html: string; css: string; js?: string }[] = []
   360→   360→      let siteName = ''
   361→   361→
   362→   362→      for (let i = 0; i < PAGE_ORDER.length; i++) {
   363→   363→        if (cancelled || cancelRef.current) return
   364→   364→        const pageInfo = PAGE_ORDER[i]
   365→   365→
   366→   366→        setPageStates(prev => prev.map(p => p.id === pageInfo.id ? { ...p, status: 'generating' } : p))
   367→   367→        setGenerationStatus(`Generating ${pageInfo.name} page…`)
   368→   368→        setGenerationProgress(Math.round((i / PAGE_ORDER.length) * 100))
   369→   369→
   370→   370→        try {
   371→   371→          // POLLING APPROACH (fixes 502 from preview-proxy):
   372→   372→          // 1. POST /api/generate → returns { jobId } immediately
   373→   373→          // 2. Poll GET /api/generate?jobId=X every 2.5s until status is
   374→   374→          //    'done' or 'error'
   375→   375→          // 3. Each request is short (~100ms) so no proxy can 502 out
   376→   376→          //
   377→   377→          // The GLM-4.7 API takes 60-150s per page, but the work runs in
   378→   378→          // the background on the server. The client just polls.
   379→   379→          const startRes = await fetch('/api/generate', {
   380→   380→            method: 'POST',
   381→   381→            headers: { 'Content-Type': 'application/json' },
   382→   382→            body: JSON.stringify({
   383→   383→              prompt: builderPrompt,

   392→   392→          if (!startRes.ok) {
   393→   393→            const errBody = await startRes.json().catch(() => ({ error: `HTTP ${startRes.status}` }))
   394→   394→            throw new Error(errBody.error || `HTTP ${startRes.status}`)
   395→   395→          }
   396→   396→
   397→   397→          const startData = await startRes.json()
   398→   398→          const jobId = startData?.jobId
   399→   399→          if (!jobId) {
   400→   400→            throw new Error('Server did not return a jobId')
   401→   401→          }
   402→   402→
   403→   403→          // Poll until done/error
   404→   404→          let data: any = null
   405→   405→          let streamError: string | null = null
   406→   406→          const POLL_INTERVAL_MS = 2500
   407→   407→          const MAX_POLL_DURATION_MS = 8 * 60 * 1000 // 8 min per page max
   408→   408→          const pollStart = Date.now()
   409→   409→
   410→   410→          while (true) {
   411→   411→            if (cancelled || cancelRef.current) return
   412→   412→
   413→   413→            const elapsed = Date.now() - pollStart
   414→   414→            if (elapsed > MAX_POLL_DURATION_MS) {
   415→   415→              throw new Error(`Timed out after ${Math.round(elapsed / 1000)}s waiting for ${pageInfo.name} page`)
   416→   416→            }
   417→   417→
   418→   418→            // Wait before next poll (but not on first iteration)
   419→   419→            if (elapsed > 0) {
   420→   420→              await new Promise<void>(r => setTimeout(r, POLL_INTERVAL_MS))
   421→   421→              if (cancelled || cancelRef.current) return
   422→   422→            }
   423→   423→
   424→   424→            const statusRes = await fetch(`/api/generate?jobId=${encodeURIComponent(jobId)}`, {
   425→   425→              method: 'GET',
   426→   426→              headers: { 'Cache-Control': 'no-cache' },
   427→   427→            })
   428→   428→
   429→   429→            if (!statusRes.ok) {
   430→   430→              // 404 = job expired/gone
   431→   431→              if (statusRes.status === 404) {
   432→   432→                throw new Error('Generation job disappeared — please retry')
   433→   433→              }
   434→   434→              const errBody = await statusRes.json().catch(() => ({ error: `HTTP ${statusRes.status}` }))
   435→   435→              throw new Error(errBody.error || `Status poll HTTP ${statusRes.status}`)
   436→   436→            }
   437→   437→
   438→   438→            const status = await statusRes.json()
   439→   439→            if (cancelled || cancelRef.current) return

   446→   446→            }
   447→   447→
   448→   448→            if (status.status === 'done') {
   449→   449→              data = status
   450→   450→              break
   451→   451→            }
   452→   452→            if (status.status === 'error') {
   453→   453→              streamError = status.error || 'Generation failed'
   454→   454→              break
   455→   455→            }
   456→   456→            // else: 'pending' or 'generating' — keep polling
   457→   457→          }

   460→   460→

   534→   534→
   535→   535→  const formatTime = (s: number) => `${Math.floor(s / 60)}:${(s % 60).toString().padStart(2, '0')}`

   538→   538→
