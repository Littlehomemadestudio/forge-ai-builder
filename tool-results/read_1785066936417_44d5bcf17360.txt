    88→// ─── Animated Counter ────────────────────────────────────────────────
    89→
    90→function AnimatedCounter({ target, suffix = '', duration = 2000 }: { target: number; suffix?: string; duration?: number }) {
    91→  const [count, setCount] = useState(0)
    92→  const ref = useRef<HTMLSpanElement>(null)
    93→  const inView = useInView(ref, { once: true, margin: '-50px' })
    94→  const hasAnimated = useRef(false)
    95→
    96→  useEffect(() => {
    97→    if (!inView || hasAnimated.current) return
    98→    hasAnimated.current = true
    99→    const startTime = performance.now()
   100→
   101→    const animate = (now: number) => {
   102→      const elapsed = now - startTime
   103→      const progress = Math.min(elapsed / duration, 1)
   104→      const eased = 1 - Math.pow(1 - progress, 3)
   105→      setCount(Math.floor(target * eased))
   106→      if (progress < 1) {
   107→        requestAnimationFrame(animate)
   108→      }
   109→    }
   110→
   111→    requestAnimationFrame(animate)
   112→  }, [inView, target, duration])
   113→
   114→  return <span ref={ref}>{count.toLocaleString()}{suffix}</span>
   115→}

   254→// ─── Interactive Builder Playground ──────────────────────────────────
   255→
   256→function BuilderPlayground() {
   257→  const [placedBlocks, setPlacedBlocks] = useState<PlaygroundBlock[]>([])
   258→  const [draggedBlock, setDraggedBlock] = useState<string | null>(null)
   259→  const [dragOverIndex, setDragOverIndex] = useState<number | null>(null)
   260→  const canvasRef = useRef<HTMLDivElement>(null)
   261→
   262→  const handleDragStart = useCallback((blockId: string) => {
   263→    setDraggedBlock(blockId)
   264→  }, [])
   265→
   266→  const handleDragEnd = useCallback(() => {
   267→    setDraggedBlock(null)
   268→    setDragOverIndex(null)
   269→  }, [])
   270→
   271→  const handleCanvasDrop = useCallback((e: React.DragEvent) => {
   272→    e.preventDefault()
   273→    if (!draggedBlock) return
   274→    const block = BLOCK_TYPES.find(b => b.id === draggedBlock)
   275→    if (block) {
   276→      const newBlock = { ...block, id: `${block.id}-${Date.now()}` }
   277→      if (dragOverIndex !== null) {
   278→        setPlacedBlocks(prev => {
   279→          const newBlocks = [...prev]
   280→          newBlocks.splice(dragOverIndex, 0, newBlock)
   281→          return newBlocks
   282→        })
   283→      } else {
   284→        setPlacedBlocks(prev => [...prev, newBlock])
   285→      }
   286→    }
   287→    setDraggedBlock(null)
   288→    setDragOverIndex(null)
   289→  }, [draggedBlock, dragOverIndex])
   290→
   291→  const handleCanvasDragOver = useCallback((e: React.DragEvent) => {
   292→    e.preventDefault()
   293→    if (!canvasRef.current) return
   294→    const rect = canvasRef.current.getBoundingClientRect()
   295→    const y = e.clientY - rect.top
   296→    const blockHeight = 50
   297→    const index = Math.floor(y / blockHeight)
   298→    setDragOverIndex(Math.min(index, placedBlocks.length))
   299→  }, [placedBlocks.length])
   300→
   301→  const handleShuffle = useCallback(() => {
   302→    setPlacedBlocks(prev => {
   303→      const shuffled = [...prev]
   304→      for (let i = shuffled.length - 1; i > 0; i--) {
   305→        const j = Math.floor(Math.random() * (i + 1))
   306→        ;[shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]]
   307→      }
   308→      return shuffled
   309→    })
   310→  }, [])
   311→
   312→  const handleClear = useCallback(() => {
   313→    setPlacedBlocks([])
   314→  }, [])
   315→
   316→  const handleRemoveBlock = useCallback((blockId: string) => {
   317→    setPlacedBlocks(prev => prev.filter(b => b.id !== blockId))
   318→  }, [])
   319→
   320→  return (
   321→    <div className="grid grid-cols-1 md:grid-cols-2 gap-4 md:gap-6">
   322→      {/* Block Palette */}
   323→      <div>
   324→        <p className="text-xs text-gray-400 mb-3 font-medium uppercase tracking-wider">Drag blocks to build</p>
   325→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   326→          {BLOCK_TYPES.map((block) => {
   327→            const Icon = block.icon
   328→            return (
   329→              <motion.div
   330→                key={block.id}
   331→                draggable
   332→                onDragStart={() => handleDragStart(block.id)}
   333→                onDragEnd={handleDragEnd}
   334→                whileHover={{ scale: 1.05, y: -2 }}
   335→                whileTap={{ scale: 0.95 }}
   336→                className="flex items-center gap-2 px-3 py-2.5 rounded-lg cursor-grab active:cursor-grabbing transition-shadow"
   337→                style={{
   338→                  background: `${block.color}15`,
   339→                  border: `1px solid ${block.color}30`,
   340→                }}
   341→              >
   342→                <div
   343→                  className="w-6 h-6 rounded flex items-center justify-center flex-shrink-0"
   344→                  style={{ background: block.bgGradient }}
   345→                >
   346→                  <Icon className="w-3.5 h-3.5 text-white" />
   347→                </div>
   348→                <span className="text-xs font-medium text-white/90">{block.label}</span>
   349→                <GripVertical className="w-3 h-3 text-white/30 ml-auto" />
   350→              </motion.div>
   351→            )
   352→          })}
   353→        </div>
   354→        <div className="flex gap-2 mt-4">
   355→          <Button
   356→            size="sm"
   357→            variant="outline"
   358→            onClick={handleShuffle}
   359→            disabled={placedBlocks.length < 2}
   360→            className="text-xs border-white/10 bg-white/5 text-white/70 hover:bg-white/10 hover:text-white"
   361→          >
   362→            <Shuffle className="w-3 h-3 mr-1" /> Shuffle
   363→          </Button>
   364→          <Button
   365→            size="sm"
   366→            variant="outline"
   367→            onClick={handleClear}
   368→            disabled={placedBlocks.length === 0}
   369→            className="text-xs border-white/10 bg-white/5 text-white/70 hover:bg-white/10 hover:text-white"
   370→          >
   371→            <Trash2 className="w-3 h-3 mr-1" /> Clear
   372→          </Button>
   373→        </div>
   374→      </div>
   375→
   376→      {/* Mini Canvas Preview */}
   377→      <div>
   378→        <p className="text-xs text-gray-400 mb-3 font-medium uppercase tracking-wider">Live preview</p>
   379→        <div
   380→          ref={canvasRef}
   381→          onDrop={handleCanvasDrop}
   382→          onDragOver={handleCanvasDragOver}
   383→          onDragLeave={() => setDragOverIndex(null)}
   384→          className="min-h-[280px] rounded-xl border border-white/10 bg-gray-950/50 p-3 relative overflow-hidden"
   385→          style={{
   386→            backgroundImage: 'radial-gradient(circle at 1px 1px, oklch(1 0 0 / 5%) 1px, transparent 0)',
   387→            backgroundSize: '20px 20px',
   388→          }}
   389→        >
   390→          {/* Navbar mockup */}
   391→          <div className="h-5 rounded bg-white/5 mb-2 flex items-center px-2 gap-1">
   392→            <div className="w-2 h-2 rounded-full bg-rose-500/60" />
   393→            <div className="w-2 h-2 rounded-full bg-amber-500/60" />
   394→            <div className="w-2 h-2 rounded-full bg-emerald-500/60" />
   395→            <div className="flex-1" />
   396→            <div className="w-6 h-1.5 rounded bg-white/10" />
   397→            <div className="w-6 h-1.5 rounded bg-white/10" />
   398→          </div>
   399→
   400→          {/* Placed blocks */}
   401→          <AnimatePresence mode="popLayout">
   402→            {placedBlocks.map((block, index) => {
   403→              const Icon = block.icon
   404→              return (
   405→                <motion.div
   406→                  key={block.id}
   407→                  layout
   408→                  initial={{ opacity: 0, scale: 0.8, y: -10 }}
   409→                  animate={{ opacity: 1, scale: 1, y: 0 }}
   410→                  exit={{ opacity: 0, scale: 0.8, x: -50 }}
   411→                  transition={{ type: 'spring', stiffness: 300, damping: 25 }}
   412→                  className="relative group mb-1.5"
   413→                >
   414→                  {dragOverIndex === index && (
   415→                    <div className="absolute -top-1 left-0 right-0 h-0.5 bg-cyan-400 rounded-full" />
   416→                  )}
   417→                  <div
   418→                    className={`${block.height} rounded-lg flex items-center px-3 gap-2 cursor-pointer transition-all`}
   419→                    style={{
   420→                      background: block.bgGradient,
   421→                      opacity: 0.9,
   422→                    }}
   423→                    onMouseEnter={(e) => { (e.currentTarget as HTMLDivElement).style.opacity = '1' }}
   424→                    onMouseLeave={(e) => { (e.currentTarget as HTMLDivElement).style.opacity = '0.9' }}
   425→                  >
   426→                    <Icon className="w-3 h-3 text-white/80" />
   427→                    <span className="text-[10px] font-medium text-white/80">{block.label}</span>
   428→                    <button
   429→                      onClick={() => handleRemoveBlock(block.id)}
   430→                      className="ml-auto opacity-0 group-hover:opacity-100 transition-opacity w-4 h-4 rounded-full bg-black/30 flex items-center justify-center"
   431→                    >
   432→                      <span className="text-white/60 text-[8px] leading-none">✕</span>
   433→                    </button>
   434→                  </div>
   435→                </motion.div>
   436→              )
   437→            })}
   438→          </AnimatePresence>
   439→
   440→          {/* Drop indicator at end */}
   441→          {dragOverIndex === placedBlocks.length && draggedBlock && (
   442→            <div className="h-0.5 bg-cyan-400 rounded-full mb-1.5" />
   443→          )}
   444→
   445→          {/* Empty state */}
   446→          {placedBlocks.length === 0 && (
   447→            <div className="flex flex-col items-center justify-center h-48 text-gray-500">
   448→              <LayoutGrid className="w-8 h-8 mb-2 text-gray-600" />
   449→              <p className="text-xs">Drag blocks here to build your site</p>
   450→            </div>
   451→          )}
   452→
   453→          {/* Animated grid overlay */}
   454→          <div className="absolute inset-0 pointer-events-none opacity-20">
   455→            <div className="absolute top-4 right-4 w-16 h-16 rounded-full bg-cyan-500/20 animate-pulse-glow" />
   456→            <div className="absolute bottom-8 left-4 w-12 h-12 rounded-full bg-rose-500/20 animate-pulse-glow" style={{ animationDelay: '2s' }} />
   457→          </div>
   458→        </div>
   459→      </div>
   460→    </div>
   461→  )
   462→}
