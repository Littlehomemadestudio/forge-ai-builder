   363→            sendElementInfo(el);
   364→          }
   365→        }
   366→        break;

   369→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   370→        if (el && el.parentElement) {
   371→          const dir = msg.data.direction;
   372→          if (dir === 'up' && el.previousElementSibling) {
   373→            el.parentElement.insertBefore(el, el.previousElementSibling);
   374→          } else if (dir === 'down' && el.nextElementSibling) {
   375→            el.parentElement.insertBefore(el.nextElementSibling, el);
   376→          }
   377→          assignIds();
   378→          if (selectedId === msg.data.elementId) {
   379→            showSelection(selectedId);
   380→            sendElementInfo(el);
   381→          }
   382→        }
   383→        break;
   384→      }
   385→      case 'duplicate-element': {
   386→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   387→        if (el && el.parentElement) {
   388→          const clone = el.cloneNode(true);
   389→          clone.removeAttribute('data-fid');
   390→          el.parentElement.insertBefore(clone, el.nextElementSibling);
   391→          assignIds();
   392→        }
   393→        break;
   394→      }
   395→      case 'add-component': {
   396→        const parent = document.querySelector('[data-fid="' + msg.data.parentId + '"]');
   397→        if (parent) {
   398→          const temp = document.createElement('div');
   399→          temp.innerHTML = msg.data.html;
   400→          const newEl = temp.firstElementChild;
   401→          if (msg.data.position >= 0 && parent.children[msg.data.position]) {
   402→            parent.insertBefore(newEl, parent.children[msg.data.position]);
   403→          } else {
   404→            parent.appendChild(newEl);
   405→          }
   406→          assignIds();
   407→        } else {
   408→          const temp = document.createElement('div');
   409→          temp.innerHTML = msg.data.html;
   410→          document.body.appendChild(temp.firstElementChild);
   411→          assignIds();
   412→        }
   413→        break;
   414→      }
   415→      case 'remove-element': {
   416→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   417→        if (el) {
   418→          el.remove();
   419→          if (selectedId === msg.data.elementId) hideSelection();
   420→          assignIds();
   421→        }
   422→        break;
   423→      }
   424→      case 'select-element': {
   425→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   426→        if (el) { showSelection(msg.data.elementId); sendElementInfo(el); }
   427→        break;
   428→      }
   429→      case 'highlight-element': { showHover(msg.data.elementId); break; }
   430→      case 'update-html': {
   431→        document.body.innerHTML = msg.data.html;
   432→        assignIds();
   433→        createOverlays();
   434→        hideSelection();
   435→        hideHover();
   436→        break;
   437→      }
   438→      case 'toggle-grid': {
   439→        if (gridEl) gridEl.style.display = msg.data.show ? 'block' : 'none';
   440→        break;
   441→      }
   442→      case 'apply-theme': {
   443→        const root = document.documentElement;
   444→        if (msg.data.colors) {
   445→          Object.entries(msg.data.colors).forEach(([key, val]) => {
   446→            root.style.setProperty('--' + key, val, 'important');
   447→            // Also update all elements with matching colors
   448→          });
   449→        }
   450→        if (msg.data.fontFamily) {
   451→          root.style.setProperty('--font-sans', msg.data.fontFamily, 'important');
   452→          document.querySelectorAll('*').forEach(el => {
   453→            const computed = getComputedStyle(el);
   454→            if (computed.fontFamily.includes('Inter') || computed.fontFamily.includes('system-ui') || computed.fontFamily.includes('sans-serif')) {
   455→              el.style.setProperty('font-family', msg.data.fontFamily, 'important');
   456→            }
   457→          });
   458→        }
   459→        break;
   460→      }
   461→      case 'get-elements-tree': {
   462→        function buildTree(el, depth) {
   463→          const fid = el.getAttribute('data-fid');
   464→          if (!fid) return null;
   465→          const children = [];
   466→          Array.from(el.children).forEach(child => {
   467→            const c = buildTree(child, depth + 1);
   468→            if (c) children.push(c);
   469→          });
   470→          return {
   471→            id: fid,
   472→            tag: el.tagName.toLowerCase(),
   473→            className: el.className || '',
   474→            textContent: (el.textContent || '').substring(0, 50),
   475→            children: children,
   476→            depth: depth
   477→          };
   478→        }
   479→        const tree = buildTree(document.body, 0);
   480→        window.parent.postMessage({ type: 'elements-tree', data: tree }, '*');
   481→        break;
   482→      }
   483→    }
   484→  });
   485→
   486→  assignIds();
   487→  createOverlays();
   488→  window.parent.postMessage({ type: 'ready', data: null }, '*');
   489→})();
   490→`
   491→}
