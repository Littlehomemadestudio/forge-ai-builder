    33→    33→/** Find a DOM node by its escaped path fid inside the canvas root. */
    34→    34→function findByFid(root: HTMLElement | null, fid: string): HTMLElement | null {
    35→    35→  if (!root) return null
    36→    36→  const parts = fid.split('-').map(Number)
    37→    37→  let el: Element = root
    38→    38→  for (const idx of parts) {
    39→    39→    const children = Array.from(el.children).filter((c) => c.nodeType === 1)
    40→    40→    if (idx >= children.length) return null
    41→    41→    el = children[idx]
    42→    42→  }
    43→    43→  return el as HTMLElement
    44→    44→}

   235→   235→          }

   238→   238→          let data: unknown
   239→   239→          try { data = JSON.parse(dataStr) } catch { data = dataStr }
   240→   240→
   241→   241→          switch (ev) {

   243→   243→              const chunk = (data as { chunk?: string })?.chunk || ''

   245→   245→              setAiFeedback(streamingText.length > 200 ? streamingText.slice(0, 200) + '…' : streamingText || 'AI is thinking…')
   246→   246→              break
   247→   247→            }

   249→   249→              suggestion = data as typeof suggestion
   250→   250→              break consume
   251→   251→            case 'error':
   252→   252→              {
   253→   253→                const errData = data as { message?: string; error?: string }

   255→   255→                setAiFeedback(m)
   256→   256→                announce(m)
   257→   257→                return
   258→   258→              }
   259→   259→            case 'done':
   260→   260→              break consume
   261→   261→            default:
   262→   262→              break
   263→   263→          }
   264→   264→        }
   265→   265→      }
   266→   266→

   286→   286→          }
   287→   287→        }
   288→   288→      }

   398→   398→        case 'command-palette':
   399→   399→        case 'quick-command': setCmdOpen((o) => !o); break
   400→   400→        case 'zoom-in': setZoom((z) => Math.min(200, z + 10)); break
   401→   401→        case 'zoom-out': setZoom((z) => Math.max(20, z - 10)); break
   402→   402→        case 'group': case 'ungroup': announce('Grouping requires multi-select'); break
   403→   403→      }
   404→   404→    }
   405→   405→    window.addEventListener('keydown', onKey)
   406→   406→    return () => window.removeEventListener('keydown', onKey)
   407→   407→  }, [undo, redo, duplicate, remove])
   408→   408→
   409→   409→  // ── Empty-canvas / start actions ──────────────────────────────────────
   410→   410→  const onEmptyAction = useCallback((action: string) => {
   411→   411→    if (action === 'start-blank') { setHtml(STARTER); setCss(''); setHistory([STARTER]); setHistoryIndex(0); announce('Blank accessible page created') }
   412→   412→    else if (action === 'templates') navigate('builder')
   413→   413→    else if (action === 'ai') { announce('Describe your site in the AI bar below'); setAiFeedback('Tell the AI what to build, then press Go.') }
   414→   414→    else if (action === 'import') { announce('Import URL is not wired to a backend yet') }
   415→   415→    else if (action === 'paste') {
   416→   416→      const pasted = window.prompt('Paste your HTML below:')
   417→   417→      if (pasted && pasted.trim()) { setHtml(pasted.trim()); setCss(''); setHistory([pasted.trim()]); setHistoryIndex(0); announce('HTML imported') }
   418→   418→    }
   419→   419→  }, [navigate])
   420→   420→
   421→   421→  const onPublish = useCallback(() => {
   422→   422→    showToast('Publishing is not configured on this host yet.')
   423→   423→    announce('Publishing not configured')
   424→   424→  }, [showToast])
   425→   425→
   426→   426→  const onExport = useCallback(() => {
   427→   427→    const full = `<!doctype html><html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width, initial-scale=1"><title>${projectName}</title><style>${css}</style></head><body>${html}</body></html>`
   428→   428→    const blob = new Blob([full], { type: 'text/html' })
   429→   429→    const url = URL.createObjectURL(blob)
   430→   430→    const a = document.createElement('a'); a.href = url; a.download = 'index.html'; a.click()
   431→   431→    URL.revokeObjectURL(url)
   432→   432→    showToast('Exported index.html')
   433→   433→    announce('Exported index.html')
   434→   434→  }, [html, css, projectName, showToast])
   435→   435→
   436→   436→  const runCommand = useCallback((id: string) => {
   437→   437→    switch (id) {
   438→   438→      case 'undo': undo(); break
   439→   439→      case 'redo': redo(); break
   440→   440→      case 'duplicate': duplicate(); break
   441→   441→      case 'delete': remove(); break
   442→   442→      case 'add-section': commit(htmlRef.current + '<section style="padding:64px 24px"></section>', 'Add section'); break
   443→   443→      case 'add-text': commit(htmlRef.current + '<h2 style="padding:8px 0;color:#111827">New heading</h2>', 'Add text'); break
   444→   444→      case 'add-image': commit(htmlRef.current + '<img src="https://placehold.co/800x400?text=Image" alt="Placeholder image" style="max-width:100%">', 'Add image'); break
   445→   445→      case 'ai-generate': case 'ai-redesign': setCmdOpen(false); runAI(id === 'ai-generate' ? 'Generate a new section' : 'Redesign the page'); break
   446→   446→      case 'accessibility': runAudit(); break
   447→   447→      case 'publish': onPublish(); break
   448→   448→      case 'shortcuts': setShortcutsOpen(true); break
   449→   449→      case 'toggle-dark': setDarkMode((d) => !d); break
   450→   450→      case 'toggle-inspector': setInspectorOpen((o) => !o); break
   451→   451→    }
   452→   452→  }, [undo, redo, duplicate, remove, commit, runAI, runAudit, onPublish])
   453→   453→
   454→   454→  // ── Render ────────────────────────────────────────────────────────────
   455→   455→  const hasContent = html.trim().length > 0
   456→   456→  const baseFont = fontSizeScale === 'small' ? 14 : fontSizeScale === 'large' ? 18 : fontSizeScale === 'extra-large' ? 20 : 16
   457→   457→
   458→   458→  // Context-aware AI suggestions
   459→   459→  const aiSuggestions = React.useMemo(() => {
   460→   460→    const base = [{ id: 'improve', label: '✨ Improve selected' }]
   461→   461→    if (selection?.isImage) base.push({ id: 'alt', label: '✨ Add alt text' })
   462→   462→    if (selection?.isText) base.push({ id: 'rewrite', label: '✨ Rewrite text' })
   463→   463→    base.push({ id: 'contrast', label: '✨ Fix contrast' })
   464→   464→    return base
   465→   465→  }, [selection])
   466→   466→
   467→   467→  return (
   468→   468→    <div className={`ve-shell${darkMode ? ' ve-dark' : ''}`} style={{ height: '100vh', width: '100vw', display: 'flex', flexDirection: 'column', overflow: 'hidden', fontSize: baseFont }}>
   469→   469→      <LiveRegion />
   470→   470→
   471→   471→      {/* Saving indicator */}
   472→   472→      {saving && (
   473→   473→        <div aria-live="polite" style={{ position: 'fixed', top: 60, right: 16, zIndex: 600, background: COLORS.primary, color: '#FFF', padding: '6px 14px', borderRadius: RADIUS.lg, fontSize: 12, fontWeight: 500, display: 'flex', alignItems: 'center', gap: 6 }}>
   474→   474→          <span className="ve-spinner" style={{ width: 12, height: 12, border: '2px solid rgba(255,255,255,0.3)', borderTopColor: '#FFF', borderRadius: '50%', display: 'inline-block' }} />
   475→   475→          Saving…
   476→   476→        </div>
   477→   477→      )}
   478→   478→
   479→   479→      <TopNav
   480→   480→        projectName={projectName}
   481→   481→        onProjectNameChange={setProjectName}
   482→   482→        canUndo={historyIndex > 0}
   483→   483→        canRedo={historyIndex < history.length - 1}
   484→   484→        onUndo={undo}
   485→   485→        onRedo={redo}
   486→   486→        onBack={() => navigate('dashboard')}
   487→   487→        device={device}
   488→   488→        onDeviceChange={setDevice}
   489→   489→        zoom={zoom}
   490→   490→        onZoomIn={() => setZoom((z) => Math.min(200, z + 10))}
   491→   491→        onZoomOut={() => setZoom((z) => Math.max(20, z - 10))}
   492→   492→        onFit={() => setZoom(70)}
   493→   493→        onCommandPalette={() => setCmdOpen(true)}
   494→   494→        onShortcuts={() => setShortcutsOpen(true)}
   495→   495→        onPublish={onPublish}
   496→   496→        onToggleToolbar={() => setToolbarVisible((v) => !v)}
   497→   497→      />
   498→   498→
   499→   499→      <div style={{ flex: 1, display: 'flex', minHeight: 0 }}>
   500→   500→        {toolbarVisible && (
   501→   501→          <IconToolbar
   502→   502→            active={activeTool}
   503→   503→            onSelect={(id) => {
   504→   504→              if (id === activeTool) {
   505→   505→                setPanelOpen((p) => !p)
   506→   506→              } else {
   507→   507→                setActiveTool(id)
   508→   508→                setPanelOpen(true)
   509→   509→              }
   510→   510→              if (id === 'ai') announce('AI: describe a request in the assistant bar below')
   511→   511→            }}
   512→   512→          />
   513→   513→        )}
   514→   514→
   515→   515→        {toolbarVisible && panelOpen && (
   516→   516→          <ToolPanel
   517→   517→            activeTool={activeTool}
   518→   518→            onInsert={onInsert}
   519→   519→            darkMode={darkMode}
   520→   520→            htmlContent={html}
   521→   521→          />
   522→   522→        )}
   523→   523→
   524→   524→        <div style={{ flex: 1, display: 'flex', flexDirection: 'column', minWidth: 0 }}>
   525→   525→          <div style={{ flex: 1, display: 'flex', position: 'relative', minHeight: 0 }}>
   526→   526→            {hasContent ? (
   527→   527→              <Canvas
   528→   528→                html={html}
   529→   529→                css={css}
   530→   530→                device={device}
   531→   531→                zoom={zoom}
   532→   532→                selectedFid={selection?.fid || null}
   533→   533→                spaceHeld={spaceHeld}
   534→   534→                onSelect={setSelection}
   535→   535→                onChangeHtml={commit}
   536→   536→                hasContent={hasContent}
   537→   537→                onContentReady={(el) => { contentRootRef.current = el }}
   538→   538→              />
   539→   539→            ) : (
   540→   540→              <div style={{ flex: 1, background: darkMode ? DARK_COLORS.background : COLORS.background, overflow: 'auto' }}>
   541→   541→                <EmptyCanvas onAction={onEmptyAction} />
   542→   542→              </div>
   543→   543→            )}
   544→   544→
   545→   545→            <FloatingSelectionBar
   546→   546→              selection={selection}
   547→   547→              onDuplicate={duplicate}
   548→   548→              onDelete={remove}
   549→   549→              onAI={() => runAI('Improve this element')}
   550→   550→              onBold={bold}
   551→   551→              onAlign={align}
   552→   552→            />
   553→   553→          </div>
   554→   554→
   555→   555→          <AIAssistantBar
   556→   556→            onSubmit={(prompt) => runAI(prompt)}
   557→   557→            isBusy={aiBusy}
   558→   558→            onStop={stopAI}
   559→   559→            onRegenerate={() => runAI('Improve this element')}
   560→   560→            lastResponse={aiFeedback}
   561→   561→            suggestions={aiSuggestions}
   562→   562→          />
   563→   563→          {aiFeedback && (
   564→   564→            <div role="status" aria-live="polite" style={{ padding: '8px 16px', background: darkMode ? DARK_COLORS.primaryLight : COLORS.primaryLight, color: darkMode ? DARK_COLORS.primary : COLORS.primary, fontSize: 13, borderTop: `1px solid ${darkMode ? DARK_COLORS.selectionLight : COLORS.selectionLight}` }}>
   565→   565→              ✨ {aiFeedback}
   566→   566→            </div>
   567→   567→          )}
   568→   568→          {auditResults.length > 0 && (
   569→   569→            <div role="status" style={{ padding: '8px 16px', background: darkMode ? DARK_COLORS.warningLight : COLORS.warningLight, color: '#92400E', fontSize: 13, borderTop: `1px solid ${darkMode ? DARK_COLORS.warning : COLORS.warning}` }}>
   570→   570→              <strong style={{ fontWeight: 600 }}>A11y audit:</strong> {auditResults.map((r) => <span key={r} style={{ display: 'inline-block', marginRight: 12 }}>• {r}</span>)}
   571→   571→            </div>
   572→   572→          )}
   573→   573→        </div>
   574→   574→
   575→   575→        {inspectorOpen && (
   576→   576→          <Inspector
   577→   577→            selection={selection}
   578→   578→            onApply={commitNode}
   579→   579→            onSelectText={onSelectText}
   580→   580→            onDeleteSelection={remove}
   581→   581→            onDuplicateSelection={duplicate}
   582→   582→            onRunAccessibilityAudit={runAudit}
   583→   583→            onFixAccessibility={fixAccessibility}
   584→   584→            onExport={onExport}
   585→   585→            onAI={(prompt) => runAI(prompt)}
   586→   586→            darkMode={darkMode}
   587→   587→            onToggleDarkMode={() => setDarkMode((d) => !d)}
   588→   588→            onToggleInspector={() => setInspectorOpen(false)}
   589→   589→          />
   590→   590→        )}
   591→   591→      </div>
   592→   592→
   593→   593→      <CommandPalette open={cmdOpen} onClose={() => setCmdOpen(false)} onRun={runCommand} />
   594→   594→      <ShortcutsHelp open={shortcutsOpen} onClose={() => setShortcutsOpen(false)} />
   595→   595→
   596→   596→      {toast && (
   597→   597→        <div role="status" style={{ position: 'fixed', bottom: 80, left: '50%', transform: 'translateX(-50%)', background: darkMode ? DARK_COLORS.text : COLORS.text, color: '#FFF', padding: '10px 18px', borderRadius: RADIUS.lg, boxShadow: SHADOWS.lg, zIndex: 500, fontSize: 14, fontWeight: 500 }}>
   598→   598→          {toast}
   599→   599→        </div>
   600→   600→      )}
   601→   601→    </div>
   602→   602→  )
   603→   603→}
