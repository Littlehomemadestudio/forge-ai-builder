    33→/** Find a DOM node by its escaped path fid inside the canvas root. */
    34→function findByFid(root: HTMLElement | null, fid: string): HTMLElement | null {
    35→  if (!root) return null
    36→  const parts = fid.split('-').map(Number)
    37→  let el: Element = root
    38→  for (const idx of parts) {
    39→    const children = Array.from(el.children).filter((c) => c.nodeType === 1)
    40→    if (idx >= children.length) return null
    41→    el = children[idx]
    42→  }
    43→  return el as HTMLElement
    44→}

   219→      const res = await fetch('/api/editor-suggest', {
   220→        method: 'POST',
   221→        headers: { 'Content-Type': 'application/json' },
   222→        body: JSON.stringify({
   223→          elementTag: node?.tagName.toLowerCase() || 'body',

   225→          computedStyles: {},
   226→          siteContext: prompt || 'professional website',

   236→        return
   237→      }
   238→
   239→      // Server streams SSE: delta (live) → result → done/error
   240→      const reader = res.body!.getReader()
   241→      const decoder = new TextDecoder()
   242→      let buffer = ''
   243→      let suggestion: { suggestions?: Array<{ description?: string; newHtml?: string }> } | null = null
   244→      let streamingText = ''
   245→
   246→      consume: while (true) {
   247→        const { done, value } = await reader.read()
   248→        if (done) break
   249→        buffer += decoder.decode(value, { stream: true })
   250→
   251→        let delim: RegExpMatchArray | null
   252→        while ((delim = buffer.match(/\r?\n\r?\n/)) !== null) {
   253→          const rawEvent = buffer.slice(0, delim.index!)
   254→          buffer = buffer.slice(delim.index! + delim[0].length)
   255→
   256→          let ev = 'message'
   257→          const dataLines: string[] = []
   258→          for (const line of rawEvent.split(/\r?\n/)) {
   259→            if (line.startsWith('event:')) ev = line.slice(6).trim()
   260→            else if (line.startsWith('data:')) dataLines.push(line.slice(5).trim())
   261→          }
   262→          if (dataLines.length === 0) continue
   263→          const dataStr = dataLines.join('\n')
   264→          let data: unknown
   265→          try { data = JSON.parse(dataStr) } catch { data = dataStr }
   266→
   267→          switch (ev) {
   268→            case 'delta': {
   269→              const chunk = (data as { chunk?: string })?.chunk || ''
   270→              streamingText += chunk
   271→              setAiFeedback(streamingText.length > 200 ? streamingText.slice(0, 200) + '…' : streamingText || 'AI is thinking…')
   272→              break
   273→            }
   274→            case 'result':
   275→              suggestion = data as typeof suggestion
   276→              break consume
   277→            case 'error':
   278→              {
   279→                const errData = data as { message?: string; error?: string }
   280→                const m = errData?.message || errData?.error || 'AI failed.'
   281→                setAiFeedback(m)
   282→                announce(m)
   283→                return
   284→              }
   285→            case 'done':
   286→              break consume
   287→            default:
   288→              break
   289→          }
   290→        }
   291→      }
   292→
   293→      // Apply AI-generated HTML if present
   294→      if (suggestion?.suggestions?.[0]?.newHtml) {
   295→        const newHtml = suggestion.suggestions[0].newHtml
   296→        if (node && node.parentElement && node.parentElement !== contentRootRef.current) {
   297→          // Replace selected element
   298→          const temp = document.createElement('div')
   299→          temp.innerHTML = newHtml
   300→          const replacement = temp.firstElementChild as HTMLElement | null
   301→          if (replacement) {
   302→            node.replaceWith(replacement)
   303→            const root = contentRootRef.current
   304→            if (root) commit(root.innerHTML, 'AI: replace element')
   305→          }
   306→        } else {
   307→          // Append to canvas
   308→          const root = contentRootRef.current
   309→          if (root) {
   310→            root.innerHTML += newHtml
   311→            commit(root.innerHTML, 'AI: append content')
   312→          }
   313→        }
   314→      }
   315→
   316→      const msg = suggestion?.suggestions?.[0]?.description || 'AI responded.'
   317→      setAiFeedback(msg)
   318→      announce(msg)
   319→    } catch (err: unknown) {
   320→      if (err instanceof DOMException && err.name === 'AbortError') {
   321→        setAiFeedback('AI request cancelled.')
   322→        announce('AI request cancelled')
   323→      } else {

   326→      }

   452→  // ── Render ────────────────────────────────────────────────────────────
   453→  const hasContent = html.trim().length > 0
   454→  const baseFont = fontSizeScale === 'small' ? 14 : fontSizeScale === 'large' ? 18 : fontSizeScale === 'extra-large' ? 20 : 16
   455→
   456→  // Context-aware AI suggestions
   457→  const aiSuggestions = React.useMemo(() => {
   458→    const base = [{ id: 'improve', label: '✨ Improve selected' }]
   459→    if (selection?.isImage) base.push({ id: 'alt', label: '✨ Add alt text' })
   460→    if (selection?.isText) base.push({ id: 'rewrite', label: '✨ Rewrite text' })
   461→    base.push({ id: 'contrast', label: '✨ Fix contrast' })
   462→    return base
   463→  }, [selection])
   464→
   465→  return (
   466→    <div className={`ve-shell${darkMode ? ' ve-dark' : ''}`} style={{ height: '100vh', width: '100vw', display: 'flex', flexDirection: 'column', overflow: 'hidden', fontSize: baseFont }}>
   467→      <LiveRegion />
   468→
   469→      {/* Saving indicator */}
   470→      {saving && (
   471→        <div aria-live="polite" style={{ position: 'fixed', top: 60, right: 16, zIndex: 600, background: COLORS.primary, color: '#FFF', padding: '6px 14px', borderRadius: RADIUS.lg, fontSize: 12, fontWeight: 500, display: 'flex', alignItems: 'center', gap: 6 }}>
   472→          <span className="ve-spinner" style={{ width: 12, height: 12, border: '2px solid rgba(255,255,255,0.3)', borderTopColor: '#FFF', borderRadius: '50%', display: 'inline-block' }} />
   473→          Saving…
   474→        </div>
   475→      )}
   476→
   477→      <TopNav
   478→        projectName={projectName}
   479→        onProjectNameChange={setProjectName}
   480→        canUndo={historyIndex > 0}
   481→        canRedo={historyIndex < history.length - 1}
   482→        onUndo={undo}
   483→        onRedo={redo}
   484→        onBack={() => navigate('dashboard')}
   485→        device={device}
   486→        onDeviceChange={setDevice}
   487→        zoom={zoom}
   488→        onZoomIn={() => setZoom((z) => Math.min(200, z + 10))}
   489→        onZoomOut={() => setZoom((z) => Math.max(20, z - 10))}
   490→        onFit={() => setZoom(70)}
   491→        onCommandPalette={() => setCmdOpen(true)}
   492→        onShortcuts={() => setShortcutsOpen(true)}
   493→        onPublish={onPublish}
   494→        onToggleToolbar={() => setToolbarVisible((v) => !v)}
   495→      />
   496→
   497→      <div style={{ flex: 1, display: 'flex', minHeight: 0 }}>
   498→        {toolbarVisible && (
   499→          <IconToolbar
   500→            active={activeTool}
   501→            onSelect={(id) => {
   502→              if (id === activeTool) {
   503→                setPanelOpen((p) => !p)
   504→              } else {
   505→                setActiveTool(id)
   506→                setPanelOpen(true)
   507→              }
   508→              if (id === 'ai') announce('AI: describe a request in the assistant bar below')
   509→            }}
   510→          />
   511→        )}
   512→
   513→        {toolbarVisible && panelOpen && (
   514→          <ToolPanel
   515→            activeTool={activeTool}
   516→            onInsert={onInsert}
   517→            darkMode={darkMode}
   518→            htmlContent={html}
   519→          />
   520→        )}
   521→
   522→        <div style={{ flex: 1, display: 'flex', flexDirection: 'column', minWidth: 0 }}>
   523→          <div style={{ flex: 1, display: 'flex', position: 'relative', minHeight: 0 }}>
   524→            {hasContent ? (
   525→              <Canvas
   526→                html={html}
   527→                css={css}
   528→                device={device}
   529→                zoom={zoom}
   530→                selectedFid={selection?.fid || null}
   531→                spaceHeld={spaceHeld}
   532→                onSelect={setSelection}
   533→                onChangeHtml={commit}
   534→                hasContent={hasContent}
   535→                onContentReady={(el) => { contentRootRef.current = el }}
   536→              />
   537→            ) : (
   538→              <div style={{ flex: 1, background: darkMode ? DARK_COLORS.background : COLORS.background, overflow: 'auto' }}>
   539→                <EmptyCanvas onAction={onEmptyAction} />
   540→              </div>
   541→            )}
   542→
   543→            <FloatingSelectionBar
   544→              selection={selection}
   545→              onDuplicate={duplicate}
   546→              onDelete={remove}
   547→              onAI={() => runAI('Improve this element')}
   548→              onBold={bold}
   549→              onAlign={align}
   550→            />
   551→          </div>
   552→
   553→          <AIAssistantBar
   554→            onSubmit={(prompt) => runAI(prompt)}
   555→            isBusy={aiBusy}
   556→            onStop={stopAI}
   557→            onRegenerate={() => runAI('Improve this element')}
   558→            lastResponse={aiFeedback}
   559→            suggestions={aiSuggestions}
   560→          />
   561→          {aiFeedback && (
   562→            <div role="status" aria-live="polite" style={{ padding: '8px 16px', background: darkMode ? DARK_COLORS.primaryLight : COLORS.primaryLight, color: darkMode ? DARK_COLORS.primary : COLORS.primary, fontSize: 13, borderTop: `1px solid ${darkMode ? DARK_COLORS.selectionLight : COLORS.selectionLight}` }}>
   563→              ✨ {aiFeedback}
   564→            </div>
   565→          )}
   566→          {auditResults.length > 0 && (
   567→            <div role="status" style={{ padding: '8px 16px', background: darkMode ? DARK_COLORS.warningLight : COLORS.warningLight, color: '#92400E', fontSize: 13, borderTop: `1px solid ${darkMode ? DARK_COLORS.warning : COLORS.warning}` }}>
   568→              <strong style={{ fontWeight: 600 }}>A11y audit:</strong> {auditResults.map((r) => <span key={r} style={{ display: 'inline-block', marginRight: 12 }}>• {r}</span>)}
   569→            </div>
   570→          )}
   571→        </div>
   572→
   573→        {inspectorOpen && (
   574→          <Inspector
   575→            selection={selection}
   576→            onApply={commitNode}
   577→            onSelectText={onSelectText}
   578→            onDeleteSelection={remove}
   579→            onDuplicateSelection={duplicate}
   580→            onRunAccessibilityAudit={runAudit}
   581→            onFixAccessibility={fixAccessibility}
   582→            onExport={onExport}
   583→            onAI={(prompt) => runAI(prompt)}
   584→            darkMode={darkMode}
   585→            onToggleDarkMode={() => setDarkMode((d) => !d)}
   586→            onToggleInspector={() => setInspectorOpen(false)}
   587→          />
   588→        )}
   589→      </div>
   590→
   591→      <CommandPalette open={cmdOpen} onClose={() => setCmdOpen(false)} onRun={runCommand} />
   592→      <ShortcutsHelp open={shortcutsOpen} onClose={() => setShortcutsOpen(false)} />
   593→
   594→      {toast && (
   595→        <div role="status" style={{ position: 'fixed', bottom: 80, left: '50%', transform: 'translateX(-50%)', background: darkMode ? DARK_COLORS.text : COLORS.text, color: '#FFF', padding: '10px 18px', borderRadius: RADIUS.lg, boxShadow: SHADOWS.lg, zIndex: 500, fontSize: 14, fontWeight: 500 }}>
   596→          {toast}
   597→        </div>
   598→      )}
   599→    </div>
   600→  )
   601→}
