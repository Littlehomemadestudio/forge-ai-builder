    86→    86→/** Find a DOM node by its path-based fid (e.g. "0-1-2") inside root. */
    87→    87→export function findByFid(root: HTMLElement | null, fid: string): HTMLElement | null {
    88→    88→  if (!root) return null
    89→    89→  const parts = fid.split('-').map(Number)
    90→    90→  let el: Element = root
    91→    91→  for (const idx of parts) {
    92→    92→    const children = Array.from(el.children).filter((c) => c.nodeType === 1)
    93→    93→    if (idx >= children.length) return null
    94→    94→    el = children[idx]
    95→    95→  }
    96→    96→  return el as HTMLElement
    97→    97→}

   100→   100→// buildFid — stable path id from element to root
   101→   101→// ═══════════════════════════════════════════════════════════════════════════
   102→   102→
   103→   103→function buildFid(el: HTMLElement, root: HTMLElement): string {
   104→   104→  const parts: string[] = []
   105→   105→  let node: HTMLElement | null = el
   106→   106→  while (node && node !== root) {
   107→   107→    const parent = node.parentElement
   108→   108→    let index = 0
   109→   109→    if (parent) {
   110→   110→      for (let sib: ChildNode | null = node.previousSibling; sib; sib = sib.previousSibling) {
   111→   111→        if (sib.nodeType === 1) index++
   112→   112→      }
   113→   113→    }
   114→   114→    parts.unshift(String(index))
   115→   115→    node = parent
   116→   116→  }
   117→   117→  return parts.join('-')
   118→   118→}

   189→   189→// Resize Handles Overlay (React component for more control)
   190→   190→// ═══════════════════════════════════════════════════════════════════════════
   191→   191→
   192→   192→interface HandlePosition {
   193→   193→  key: string
   194→   194→  style: React.CSSProperties
   195→   195→  cursor: string
   196→   196→}
   197→   197→
   198→   198→function useResizeHandles(rect: DOMRect | null, zoom: number): HandlePosition[] {
   199→   199→  return useMemo(() => {
   200→   200→    if (!rect || rect.width === 0 || rect.height === 0) return []
   201→   201→
   202→   202→    const HANDLE_SIZE = 8
   203→   203→    const half = HANDLE_SIZE / 2
   204→   204→    // Handles are positioned relative to the selection rect
   205→   205→    const top = 0
   206→   206→    const left = 0
   207→   207→    const right = rect.width
   208→   208→    const bottom = rect.height
   209→   209→    const midX = rect.width / 2
   210→   210→    const midY = rect.height / 2
   211→   211→
   212→   212→    return [
   213→   213→      { key: 'nw', cursor: 'nwse-resize', style: { top: top - half, left: left - half } },
   214→   214→      { key: 'n', cursor: 'ns-resize', style: { top: top - half, left: midX - half } },
   215→   215→      { key: 'ne', cursor: 'nesw-resize', style: { top: top - half, left: right - half } },
   216→   216→      { key: 'w', cursor: 'ew-resize', style: { top: midY - half, left: left - half } },
   217→   217→      { key: 'e', cursor: 'ew-resize', style: { top: midY - half, left: right - half } },
   218→   218→      { key: 'sw', cursor: 'nesw-resize', style: { top: bottom - half, left: left - half } },
   219→   219→      { key: 's', cursor: 'ns-resize', style: { top: bottom - half, left: midX - half } },
   220→   220→      { key: 'se', cursor: 'nwse-resize', style: { top: bottom - half, left: right - half } },
   221→   221→    ].map((h) => ({
   222→   222→      ...h,
   223→   223→      style: {
   224→   224→        ...h.style,
   225→   225→        position: 'absolute' as const,
   226→   226→        width: HANDLE_SIZE,
   227→   227→        height: HANDLE_SIZE,
   228→   228→        background: '#FFFFFF',
   229→   229→        border: '1.5px solid #3B82F6',
   230→   230→        borderRadius: 1,
   231→   231→        zIndex: Z_INDEX.selection + 2,
   232→   232→        pointerEvents: 'auto' as const,
   233→   233→        cursor: h.cursor,
   234→   234→        // Subtle shadow for visibility on any background
   235→   235→        boxShadow: '0 1px 3px rgba(0,0,0,0.15)',
   236→   236→      },
   237→   237→    }))
   238→   238→  }, [rect, zoom])
   239→   239→}

   245→   245→export function Canvas(p: CanvasProps) {
   246→   246→  const theme = useEditorTheme()
   247→   247→  const { reduceMotion } = useAccessibility()
   248→   248→  const isDark = theme.mode === 'dark'
   249→   249→
   250→   250→  // ── Refs ──────────────────────────────────────────────────────────────
   251→   251→  const vpRef = useRef<HTMLDivElement>(null)
   252→   252→  const contentRef = useRef<HTMLDivElement>(null)
   253→   253→  const panning = useRef(false)
   254→   254→  const [isPanning, setIsPanning] = useState(false)

   256→   256→  const selToken = useRef<string | null>(null)
   257→   257→  const [selRect, setSelRect] = useState<DOMRect | null>(null)
   258→   258→  const [contentRect, setContentRect] = useState<DOMRect | null>(null)
   259→   259→  const [hoveredFid, setHoveredFid] = useState<string | null>(null)
   260→   260→  const hoveredFidRef = useRef<string | null>(null)
   261→   261→  const [showGrid, setShowGrid] = useState(true)
   262→   262→
   263→   263→  const width = DEVICE_WIDTH[p.device]
   264→   264→  const bezel = DEVICE_BEZEL[p.device]
   265→   265→
   266→   266→  // ── Theme-derived colors ──────────────────────────────────────────────
   267→   267→  const colors = isDark ? DARK_COLORS : LIGHT_COLORS
   268→   268→  const canvasBg = colors.canvas.background
   269→   269→  const canvasDot = colors.canvas.dot
   270→   270→  const canvasFrame = colors.canvas.frame
   271→   271→  const artboardBg = colors.canvas.artboard
   272→   272→  const selectionColor = colors.selection.fill
   273→   273→  const handleFill = colors.selection.handle
   274→   274→  const borderColor = colors.border.default
   275→   275→  const textTertiary = colors.onSurface.tertiary
   276→   276→
   277→   277→  // ── Editor CSS (theme-aware) ──────────────────────────────────────────
   278→   278→  const editorCss = useMemo(
   279→   279→    () => buildEditorCss(selectionColor, handleFill),
   280→   280→    [selectionColor, handleFill]
   281→   281→  )
   282→   282→  const fullCss = editorCss + (p.css || '')
   283→   283→
   284→   284→  // ── Dot grid background ──────────────────────────────────────────────
   285→   285→  const gridStyle = useMemo(
   286→   286→    () => ({
   287→   287→      backgroundImage: `
   288→   288→        radial-gradient(circle ${GRID.dotSize}px at center, ${canvasDot} 0.8px, transparent 0.8px),
   289→   289→        radial-gradient(circle ${GRID.dotSize * 1.5}px at center, ${canvasDot} 0.4px, transparent 0.4px)
   290→   290→      `,
   291→   291→      backgroundSize: `${GRID.minorGrid}px ${GRID.minorGrid}px, ${GRID.majorGrid}px ${GRID.majorGrid}px`,
   292→   292→    }),
   293→   293→    [canvasDot]
   294→   294→  )

   296→   296→  // ── Resize handles ───────────────────────────────────────────────────
   297→   297→  const handles = useResizeHandles(selRect, p.zoom)
   298→   298→
   299→   299→  // ── Selection highlight sync ─────────────────────────────────────────
   300→   300→  useEffect(() => {
   301→   301→    const root = contentRef.current
   302→   302→    if (!root) return
   303→   303→
   304→   304→    // Clear previous
   305→   305→    if (selToken.current) {
   306→   306→      const prev = root.querySelector(`[data-ve-sel="${selToken.current}"]`)
   307→   307→      if (prev) {
   308→   308→        prev.classList.remove('ve-selected')
   309→   309→        prev.removeAttribute('data-ve-sel')
   310→   310→      }
   311→   311→      selToken.current = null
   312→   312→      setSelRect(null)
   313→   313→      setContentRect(null)
   314→   314→    }
   315→   315→
   316→   316→    if (!p.selectedFid) return

   320→   320→      el.classList.add('ve-selected')
   321→   321→      el.setAttribute('data-ve-sel', p.selectedFid)
   322→   322→      selToken.current = p.selectedFid
   323→   323→
   324→   324→      const rect = el.getBoundingClientRect()
   325→   325→      setSelRect(rect)
   326→   326→      const cRect = contentRef.current?.getBoundingClientRect() || null
   327→   327→      setContentRect(cRect)
   328→   328→
   329→   329→      // Scroll to selected element if off-screen
   330→   330→      const vp = vpRef.current
   331→   331→      if (vp) {
   332→   332→        const vpRect = vp.getBoundingClientRect()
   333→   333→        const margin = 80
   334→   334→        if (
   335→   335→          rect.top < vpRect.top + margin ||
   336→   336→          rect.bottom > vpRect.bottom - margin ||
   337→   337→          rect.left < vpRect.left + margin ||
   338→   338→          rect.right > vpRect.right - margin
   339→   339→        ) {
   340→   340→          el.scrollIntoView({
   341→   341→            behavior: reduceMotion ? 'auto' : 'smooth',
   342→   342→            block: 'nearest',
   343→   343→            inline: 'nearest',
   344→   344→          })
   345→   345→        }
   346→   346→      }
   347→   347→    }
   348→   348→  }, [p.selectedFid, p.html, reduceMotion])

   358→   358→    if (!root) return

   360→   360→    elements.forEach((el) => {
   361→   361→      ;(el as HTMLElement).classList.add('ve-hover-target')
   362→   362→    })
   363→   363→    return () => {
   364→   364→      elements.forEach((el) => {
   365→   365→        ;(el as HTMLElement).classList.remove('ve-hover-target')
   366→   366→      })
   367→   367→    }
   368→   368→  }, [p.html])
   369→   369→
   370→   370→  // ── Track hover for dashed indicator ─────────────────────────────────

   372→   372→    (e: React.MouseEvent) => {
   373→   373→      if (panning.current) return
   374→   374→      const root = contentRef.current
   375→   375→      if (!root) return
   376→   376→
   377→   377→      const target = e.target as HTMLElement
   378→   378→      const el = target.closest(SELECTABLE) as HTMLElement | null
   379→   379→
   380→   380→      // Clear previous hover
   381→   381→      if (hoveredFidRef.current) {
   382→   382→        const prev = findByFid(root, hoveredFidRef.current)
   383→   383→        if (prev && prev !== el) {
   384→   384→          prev.classList.remove('ve-hovered')
   385→   385→        }
   386→   386→      }
   387→   387→
   388→   388→      if (el && root.contains(el) && el.classList.contains('ve-hover-target')) {
   389→   389→        const fid = buildFid(el, root)
   390→   390→        if (fid !== p.selectedFid) {
   391→   391→          el.classList.add('ve-hovered')
   392→   392→          hoveredFidRef.current = fid
   393→   393→          setHoveredFid(fid)
   394→   394→        } else {
   395→   395→          hoveredFidRef.current = null
   396→   396→          setHoveredFid(null)
   397→   397→        }
   398→   398→      } else {
   399→   399→        hoveredFidRef.current = null
   400→   400→        setHoveredFid(null)
   401→   401→      }
   402→   402→    },
   403→   403→    [p.selectedFid]
   404→   404→  )
   405→   405→
   406→   406→  const handleMouseLeave = useCallback(() => {
   407→   407→    const root = contentRef.current
   408→   408→    if (root && hoveredFidRef.current) {
   409→   409→      const prev = findByFid(root, hoveredFidRef.current)
   410→   410→      if (prev) prev.classList.remove('ve-hovered')
   411→   411→    }
   412→   412→    hoveredFidRef.current = null
   413→   413→    setHoveredFid(null)
   414→   414→  }, [])
   415→   415→
   416→   416→  // ── Read element info ────────────────────────────────────────────────
   417→   417→  const readInfo = useCallback((el: HTMLElement, root: HTMLElement): SelectionInfo | null => {
   418→   418→    const fid = buildFid(el, root)
   419→   419→    const tag = el.tagName.toLowerCase()
   420→   420→    const cs = window.getComputedStyle(el)
   421→   421→    const rect = el.getBoundingClientRect()
   422→   422→    const isSection = ['section', 'header', 'footer', 'nav', 'ul', 'li', 'form', 'div'].includes(tag)
   423→   423→    return {
   424→   424→      fid,
   425→   425→      tag,
   426→   426→      text: (el.textContent || '').trim().slice(0, 120),
   427→   427→      isImage: tag === 'img',
   428→   428→      isText: ['h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'p', 'a', 'span', 'li'].includes(tag),
   429→   429→      isButton: tag === 'button',
   430→   430→      isLink: tag === 'a',
   431→   431→      isSection,
   432→   432→      color: cs.color,
   433→   433→      bgColor:
   434→   434→        cs.backgroundColor && cs.backgroundColor !== 'rgba(0, 0, 0, 0)'
   435→   435→          ? cs.backgroundColor
   436→   436→          : undefined,
   437→   437→      fontSize: cs.fontSize,
   438→   438→      fontWeight: cs.fontWeight,
   439→   439→      rect: { left: rect.left, top: rect.top, width: rect.width, height: rect.height },
   440→   440→    }
   441→   441→  }, [])
   442→   442→
   443→   443→  // ── Click to select ──────────────────────────────────────────────────
   444→   444→  const handleClick = useCallback(
   445→   445→    (e: React.MouseEvent) => {
   446→   446→      if (panning.current) return
   447→   447→      const root = contentRef.current
   448→   448→      if (!root) return
   449→   449→      const target = e.target as HTMLElement
   450→   450→      const el = target.closest(SELECTABLE) as HTMLElement | null
   451→   451→      if (el && root.contains(el)) {
   452→   452→        p.onSelect(readInfo(el, root))
   453→   453→      } else {
   454→   454→        p.onSelect(null)
   455→   455→      }
   456→   456→    },
   457→   457→    [p.onSelect, readInfo]
   458→   458→  )

   460→   460→  // ── Drag-to-pan (Space + drag or middle mouse) ──────────────────────

   462→   462→    (e: React.PointerEvent) => {
   463→   463→      if (e.button !== 1 && !p.spaceHeld) return

   465→   465→      e.stopPropagation()
   466→   466→      panning.current = true
   467→   467→      setIsPanning(true)
   468→   468→      lastPoint.current = { x: e.clientX, y: e.clientY }
   469→   469→
   470→   470→      const vp = vpRef.current
   471→   471→
   472→   472→      const move = (ev: PointerEvent) => {
   473→   473→        if (!vp) return
   474→   474→        const dx = ev.clientX - lastPoint.current.x
   475→   475→        const dy = ev.clientY - lastPoint.current.y
   476→   476→        vp.scrollLeft -= dx
   477→   477→        vp.scrollTop -= dy
   478→   478→        lastPoint.current = { x: ev.clientX, y: ev.clientY }
   479→   479→      }
   480→   480→      const up = () => {
   481→   481→        panning.current = false
   482→   482→        setIsPanning(false)
   483→   483→        window.removeEventListener('pointermove', move)
   484→   484→        window.removeEventListener('pointerup', up)
   485→   485→      }
   486→   486→      window.addEventListener('pointermove', move)
   487→   487→      window.addEventListener('pointerup', up)
   488→   488→    },
   489→   489→    [p.spaceHeld]
   490→   490→  )
   491→   491→
   492→   492→  // ── Keyboard: nudge + delete ─────────────────────────────────────────
   493→   493→  useEffect(() => {
   494→   494→    if (!p.selectedFid) return
   495→   495→    const root = contentRef.current
   496→   496→    if (!root) return
   497→   497→    const el = findByFid(root, p.selectedFid)
   498→   498→    if (!el) return
   499→   499→
   500→   500→    const onKey = (e: KeyboardEvent) => {
   501→   501→      if (e.key.startsWith('Arrow')) {
   502→   502→        e.preventDefault()
   503→   503→        const dx = e.shiftKey ? 10 : 1
   504→   504→        const m = /translate\(([-\d.]+)px, ([-\d.]+)px\)/.exec(el.style.transform || '')
   505→   505→        const tx = m ? parseFloat(m[1]) : 0
   506→   506→        const ty = m ? parseFloat(m[2]) : 0
   507→   507→        let nx = tx,
   508→   508→          ny = ty
   509→   509→        if (e.key === 'ArrowLeft') nx -= dx
   510→   510→        else if (e.key === 'ArrowRight') nx += dx
   511→   511→        else if (e.key === 'ArrowUp') ny -= dx
   512→   512→        else if (e.key === 'ArrowDown') ny += dx
   513→   513→        el.style.transform = nx === 0 && ny === 0 ? '' : `translate(${nx}px, ${ny}px)`
   514→   514→        announce(`${p.selectedFid} nudge ${dx}px`)
   515→   515→      } else if (e.key === 'Delete' || e.key === 'Backspace') {
   516→   516→        if (el.parentElement && el.parentElement !== root) {
   517→   517→          const tag = el.tagName.toLowerCase()
   518→   518→          el.remove()
   519→   519→          announce(`Deleted ${tag}`)
   520→   520→          p.onSelect(null)
   521→   521→          setTimeout(() => p.onChangeHtml(root.innerHTML, `Delete ${tag}`), 0)
   522→   522→        } else {
   523→   523→          e.preventDefault()
   524→   524→        }
   525→   525→      }
   526→   526→    }
   527→   527→    window.addEventListener('keydown', onKey)
   528→   528→    return () => window.removeEventListener('keydown', onKey)
   529→   529→  }, [p.selectedFid, p.onChangeHtml, p.onSelect])
   530→   530→
   531→   531→  // ── Grid toggle keyboard shortcut (Ctrl+G) ───────────────────────────

   542→   542→  }, [showGrid])
   543→   543→
   544→   544→  // ── Device frame label ───────────────────────────────────────────────

   556→   556→

   574→   574→        ...(showGrid ? gridStyle : {}),
   575→   575→        // Subtle inner shadow for depth

   577→   577→      }}
   578→   578→    >
   579→   579→      {/* Scrollable world surface */}

   587→   587→          flexDirection: 'column',

   589→   589→        }}
   590→   590→      >
   591→   591→        {/* Device frame label */}
