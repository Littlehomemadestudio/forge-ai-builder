   273→function TypingAnimation() {
   274→  const prompts = [
   275→    'A minimalist portfolio for a photographer with dark theme...',
   276→    'SaaS landing page with pricing, testimonials, and hero...',
   277→    'Restaurant website with menu cards and reservation form...',
   278→    'E-commerce store for luxury watches with product grid...',
   279→  ]
   280→  const [currentPromptIndex, setCurrentPromptIndex] = useState(0)
   281→  const [displayedText, setDisplayedText] = useState('')
   282→  const isDeletingRef = useRef(false)
   283→  const [isDeletingFlag, setIsDeletingFlag] = useState(false)
   284→
   285→  useEffect(() => {
   286→    const currentPrompt = prompts[currentPromptIndex]
   287→    let timeout: NodeJS.Timeout
   288→
   289→    if (!isDeletingRef.current) {
   290→      if (displayedText.length < currentPrompt.length) {
   291→        timeout = setTimeout(() => {
   292→          setDisplayedText(currentPrompt.slice(0, displayedText.length + 1))
   293→        }, 40 + Math.random() * 30)
   294→      } else {
   295→        timeout = setTimeout(() => {
   296→          isDeletingRef.current = true
   297→          setIsDeletingFlag(true)
   298→        }, 2000)
   299→      }
   300→    } else {
   301→      if (displayedText.length > 0) {
   302→        timeout = setTimeout(() => {
   303→          setDisplayedText(displayedText.slice(0, -1))
   304→        }, 20)
   305→      } else {
   306→        timeout = setTimeout(() => {
   307→          isDeletingRef.current = false
   308→          setIsDeletingFlag(false)
   309→          setCurrentPromptIndex((prev) => (prev + 1) % prompts.length)
   310→        }, 0)
   311→      }
   312→    }
   313→
   314→    return () => clearTimeout(timeout)
   315→  }, [displayedText, currentPromptIndex, isDeletingFlag])
   316→
   317→  return (
   318→    <span className="font-mono">
   319→      {displayedText}
   320→      <span className="animate-blink-cursor text-[oklch(0.85_0.08_260)]">|</span>
   321→    </span>
   322→  )
   323→}
