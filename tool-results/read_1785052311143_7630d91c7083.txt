   329→  window.addEventListener('message', function(e) {
   330→    const msg = e.data;
   331→    if (!msg || !msg.type) return;
   332→
   333→    switch (msg.type) {
   334→      case 'apply-style': {
   335→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   336→        if (el) el.style.setProperty(msg.data.property, msg.data.value, 'important');

   338→      }

   340→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   341→        if (el) {
   342→          el.innerHTML = msg.data.content;
   343→          assignIds();
   344→          if (selectedId === msg.data.elementId) {
   345→            showSelection(selectedId);
   346→            sendElementInfo(el);
   347→          }
   348→        }
   349→        break;
   350→      }
   351→      case 'move-element': {
   352→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   353→        if (el && el.parentElement) {
   354→          const dir = msg.data.direction;
   355→          if (dir === 'up' && el.previousElementSibling) {
   356→            el.parentElement.insertBefore(el, el.previousElementSibling);
   357→          } else if (dir === 'down' && el.nextElementSibling) {
   358→            el.parentElement.insertBefore(el.nextElementSibling, el);
   359→          }
   360→          assignIds();
   361→          if (selectedId === msg.data.elementId) {
   362→            showSelection(selectedId);
   363→            sendElementInfo(el);
   364→          }
   365→        }
   366→        break;
   367→      }
   368→      case 'duplicate-element': {
   369→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   370→        if (el && el.parentElement) {
   371→          const clone = el.cloneNode(true);
   372→          clone.removeAttribute('data-fid');
   373→          el.parentElement.insertBefore(clone, el.nextElementSibling);
   374→          assignIds();
   375→        }
   376→        break;
   377→      }
   378→      case 'add-component': {
   379→        const parent = document.querySelector('[data-fid="' + msg.data.parentId + '"]');
   380→        if (parent) {
   381→          const temp = document.createElement('div');
   382→          temp.innerHTML = msg.data.html;
   383→          const newEl = temp.firstElementChild;
   384→          if (msg.data.position >= 0 && parent.children[msg.data.position]) {
   385→            parent.insertBefore(newEl, parent.children[msg.data.position]);
   386→          } else {
   387→            parent.appendChild(newEl);
   388→          }
   389→          assignIds();
   390→        } else {
   391→          const temp = document.createElement('div');
   392→          temp.innerHTML = msg.data.html;
   393→          document.body.appendChild(temp.firstElementChild);
   394→          assignIds();
   395→        }
   396→        break;
   397→      }
   398→      case 'remove-element': {
   399→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   400→        if (el) {
   401→          el.remove();
   402→          if (selectedId === msg.data.elementId) hideSelection();
   403→          assignIds();
   404→        }
   405→        break;
   406→      }
   407→      case 'select-element': {
   408→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   409→        if (el) { showSelection(msg.data.elementId); sendElementInfo(el); }
   410→        break;
   411→      }
   412→      case 'highlight-element': { showHover(msg.data.elementId); break; }
   413→      case 'update-html': {
   414→        document.body.innerHTML = msg.data.html;
   415→        assignIds();
   416→        createOverlays();
   417→        hideSelection();
   418→        hideHover();
   419→        break;
   420→      }
   421→      case 'toggle-grid': {
   422→        if (gridEl) gridEl.style.display = msg.data.show ? 'block' : 'none';
   423→        break;
   424→      }
   425→      case 'apply-theme': {
   426→        const root = document.documentElement;
   427→        if (msg.data.colors) {
   428→          Object.entries(msg.data.colors).forEach(([key, val]) => {
   429→            root.style.setProperty('--' + key, val, 'important');
   430→            // Also update all elements with matching colors
   431→          });
   432→        }
   433→        if (msg.data.fontFamily) {
   434→          root.style.setProperty('--font-sans', msg.data.fontFamily, 'important');
   435→          document.querySelectorAll('*').forEach(el => {
   436→            const computed = getComputedStyle(el);
   437→            if (computed.fontFamily.includes('Inter') || computed.fontFamily.includes('system-ui') || computed.fontFamily.includes('sans-serif')) {
   438→              el.style.setProperty('font-family', msg.data.fontFamily, 'important');
   439→            }
   440→          });
   441→        }
   442→        break;
   443→      }
   444→      case 'get-elements-tree': {
   445→        function buildTree(el, depth) {
   446→          const fid = el.getAttribute('data-fid');
   447→          if (!fid) return null;
   448→          const children = [];
   449→          Array.from(el.children).forEach(child => {
   450→            const c = buildTree(child, depth + 1);
   451→            if (c) children.push(c);
   452→          });
   453→          return {
   454→            id: fid,
   455→            tag: el.tagName.toLowerCase(),
   456→            className: el.className || '',
   457→            textContent: (el.textContent || '').substring(0, 50),
   458→            children: children,
   459→            depth: depth
   460→          };
   461→        }
   462→        const tree = buildTree(document.body, 0);
   463→        window.parent.postMessage({ type: 'elements-tree', data: tree }, '*');
   464→        break;
   465→      }
   466→    }
   467→  });
   468→
   469→  assignIds();
   470→  createOverlays();
   471→  window.parent.postMessage({ type: 'ready', data: null }, '*');
   472→})();
   473→`
   474→}
