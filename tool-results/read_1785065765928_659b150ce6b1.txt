    65→    65→function AnimatedCounter({ target, suffix = '', duration = 2000 }: { target: number; suffix?: string; duration?: number }) {
    66→    66→  const [count, setCount] = useState(0)
    67→    67→  const ref = useRef<HTMLSpanElement>(null)
    68→    68→  const inView = useInView(ref, { once: true, margin: '-50px' })
    69→    69→  const hasAnimated = useRef(false)
    70→    70→
    71→    71→  useEffect(() => {
    72→    72→    if (!inView || hasAnimated.current) return
    73→    73→    hasAnimated.current = true
    74→    74→    const start = 0
    75→    75→    const startTime = performance.now()
    76→    76→
    77→    77→    const animate = (now: number) => {
    78→    78→      const elapsed = now - startTime
    79→    79→      const progress = Math.min(elapsed / duration, 1)
    80→    80→      // Ease out cubic
    81→    81→      const eased = 1 - Math.pow(1 - progress, 3)
    82→    82→      setCount(Math.floor(start + (target - start) * eased))
    83→    83→      if (progress < 1) {
    84→    84→        requestAnimationFrame(animate)
    85→    85→      }
    86→    86→    }
    87→    87→
    88→    88→    requestAnimationFrame(animate)
    89→    89→  }, [inView, target, duration])
    90→    90→
    91→    91→  return <span ref={ref}>{count.toLocaleString()}{suffix}</span>
    92→    92→}
