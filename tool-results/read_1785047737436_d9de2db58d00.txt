    45→  // Assign unique IDs to every element
    46→  function assignIds() {

    48→    let counter = 0;
    49→    all.forEach(el => {
    50→      if (!el.getAttribute('data-fid')) {
    51→        el.setAttribute('data-fid', 'f-el-' + counter);

    53→      }
    54→    });
    55→  }
    56→
    57→  // Create overlay elements
    58→  function createOverlays() {

    60→    overlayEl.id = 'forge-hover-overlay';

    63→

    65→    selectionBox.id = 'forge-selection-box';

    74→
    75→  // Show hover overlay
    76→  function showHover(id) {
    77→    const el = document.querySelector('[data-fid="' + id + '"]');

    94→  }
    95→
    96→  // Show selection
    97→  function showSelection(id) {
    98→    const el = document.querySelector('[data-fid="' + id + '"]');
    99→    if (!el) return;

   102→    selectionBox.style.left = (rect.left - 2) + 'px';

   107→    selectionBox.style.display = 'block';

   112→    labelEl.textContent = tagName + className;
   113→    labelEl.style.top = (rect.top - 24) + 'px';
   114→    labelEl.style.left = rect.left + 'px';

   116→
   117→    selectedId = id;
   118→  }

   122→    labelEl.style.display = 'none';
   123→    selectedId = null;

   127→  function sendElementInfo(el) {
   128→    const id = el.getAttribute('data-fid');
   129→    const computed = getComputedStyle(el);
   130→    const rect = el.getBoundingClientRect();

   132→    // Collect computed styles
   133→    const styles = {};

   147→      'box-shadow','text-shadow','opacity','cursor','visibility',
   148→      'overflow','overflow-x','overflow-y','float','clear',
   149→      'flex-direction','flex-wrap','justify-content','align-items','align-self',
   150→      'flex-grow','flex-shrink','flex-basis','gap','row-gap','column-gap',
   151→      'grid-template-columns','grid-template-rows','grid-column','grid-row',
   152→      'order','transform','transform-origin','filter','backdrop-filter',
   153→      'transition','animation','object-fit','object-position',
   154→      'white-space','word-break','direction','writing-mode',
   155→      'list-style-type','list-style-position','table-layout','border-collapse',
   156→      'fill','stroke','stroke-width','mix-blend-mode','background-blend-mode',

   158→    ];
   159→    importantProps.forEach(prop => {
   160→      styles[prop] = computed.getPropertyValue(prop);
   161→    });
   162→
   163→    // Collect attributes
   164→    const attributes = {};
   165→    for (let i = 0; i < el.attributes.length; i++) {

   168→        attributes[attr.name] = attr.value;
   169→      }

   174→    const parentId = parent ? parent.getAttribute('data-fid') : null;

   177→    Array.from(el.children).forEach(child => {

   218→    if (el === overlayEl || el === selectionBox || el === labelEl) return;
   219→    const fid = el.getAttribute('data-fid');
   220→    if (fid) {
   221→      showSelection(fid);
   222→      sendElementInfo(el);
   223→    }
   224→  }, true);
   225→
   226→  // Double-click for inline text editing
   227→  document.addEventListener('dblclick', function(e) {
   228→    e.preventDefault();
   229→    e.stopPropagation();
   230→    const el = e.target;
   231→    const fid = el.getAttribute('data-fid');
   232→    if (!fid) return;
   233→
   234→    // Only allow editing on elements with direct text content
   235→    if (el.children.length === 0 || (el.children.length === 1 && el.children[0].tagName === 'BR')) {
   236→      isEditing = true;
   237→      el.contentEditable = 'true';
   238→      el.style.cursor = 'text';

   246→      sel.addRange(range);
   247→    }
   248→  }, true);
   249→
   250→  // Handle blur from inline editing
   251→  document.addEventListener('blur', function(e) {
   252→    const el = e.target;
   253→    if (el.contentEditable === 'true') {
   254→      el.contentEditable = 'false';
   255→      el.style.cursor = '';
   256→      isEditing = false;
   257→
   258→      window.parent.postMessage({
   259→        type: 'text-edited',
   260→        data: {
   261→          id: el.getAttribute('data-fid'),
   262→          content: el.innerHTML,
   263→          textContent: el.textContent
   264→        }
   265→      }, '*');
   266→    }
   267→  }, true);
   268→
   269→  // Prevent link navigation
   270→  document.addEventListener('click', function(e) {
   271→    const link = e.target.closest('a');
   272→    if (link && !isEditing) {
   273→      e.preventDefault();
   274→    }
   275→  });
   276→
   277→  // Prevent form submission
   278→  document.addEventListener('submit', function(e) {
   279→    e.preventDefault();
   280→  });
   281→
   282→  // Listen for parent messages
   283→  window.addEventListener('message', function(e) {
   284→    const msg = e.data;
   285→    if (!msg || !msg.type) return;
   286→
   287→    switch (msg.type) {
   288→      case 'apply-style': {
   289→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   290→        if (el) {
   291→          el.style.setProperty(msg.data.property, msg.data.value, 'important');
   292→        }
   293→        break;
   294→      }
   295→      case 'apply-content': {
   296→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   297→        if (el) {
   298→          el.innerHTML = msg.data.content;
   299→          assignIds();
   300→          if (selectedId === msg.data.elementId) {
   301→            showSelection(selectedId);
   302→            sendElementInfo(el);
   303→          }
   304→        }
   305→        break;
   306→      }
   307→      case 'add-component': {
   308→        const parent = document.querySelector('[data-fid="' + msg.data.parentId + '"]');
   309→        if (parent) {
   310→          const temp = document.createElement('div');
   311→          temp.innerHTML = msg.data.html;
   312→          const newEl = temp.firstElementChild;
   313→          if (msg.data.position >= 0 && parent.children[msg.data.position]) {
   314→            parent.insertBefore(newEl, parent.children[msg.data.position]);
   315→          } else {
   316→            parent.appendChild(newEl);
   317→          }
   318→          assignIds();
   319→        } else {
   320→          // Add to body
   321→          const temp = document.createElement('div');
   322→          temp.innerHTML = msg.data.html;
   323→          const newEl = temp.firstElementChild;
   324→          document.body.appendChild(newEl);
   325→          assignIds();
   326→        }
   327→        break;
   328→      }
   329→      case 'remove-element': {
   330→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   331→        if (el) {
   332→          el.remove();
   333→          if (selectedId === msg.data.elementId) hideSelection();
   334→          assignIds();
   335→        }
   336→        break;
   337→      }
   338→      case 'select-element': {
   339→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   340→        if (el) {
   341→          showSelection(msg.data.elementId);
   342→          sendElementInfo(el);
   343→        }
   344→        break;
   345→      }
   346→      case 'highlight-element': {
   347→        showHover(msg.data.elementId);
   348→        break;
   349→      }
   350→      case 'update-html': {
   351→        document.body.innerHTML = msg.data.html;
   352→        assignIds();
   353→        createOverlays();
   354→        hideSelection();
   355→        hideHover();
   356→        break;
   357→      }
   358→      case 'get-elements-tree': {
   359→        function buildTree(el, depth) {
   360→          const fid = el.getAttribute('data-fid');
   361→          if (!fid) return null;
   362→          const children = [];
   363→          // el.children is an HTMLCollection (not an array) — must convert
   364→          // before calling .forEach, otherwise this throws silently and the
   365→          // tree never gets built.
   366→          Array.from(el.children).forEach(child => {
   367→            const c = buildTree(child, depth + 1);
   368→            if (c) children.push(c);
   369→          });
   370→          return {
   371→            id: fid,
   372→            tag: el.tagName.toLowerCase(),
   373→            className: el.className || '',
   374→            textContent: (el.textContent || '').substring(0, 50),
   375→            children: children,
   376→            depth: depth
   377→          };
   378→        }
   379→        const tree = buildTree(document.body, 0);
   380→        window.parent.postMessage({ type: 'elements-tree', data: tree }, '*');
   381→        break;
   382→      }
   383→    }
   384→  });
   385→
   386→  // Initialize
   387→  assignIds();
   388→  createOverlays();
   389→
   390→  // Notify parent that iframe is ready
   391→  window.parent.postMessage({ type: 'ready', data: null }, '*');
   392→})();
   393→`
   394→}

   489→// ─── Prepare HTML for iframe (inject bridge script) ────────────────────────
   490→function prepareHTMLForIframe(html: string): string {
