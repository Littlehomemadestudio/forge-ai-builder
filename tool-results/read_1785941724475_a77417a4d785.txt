    34→// ═══════════════════════════════════════════════════════════════════════════
    35→
    36→let _announceTimer: ReturnType<typeof setTimeout> | null = null
    37→
    38→export function LiveRegion() {
    39→  return (
    40→    <div
    41→      id="ve-live-region"
    42→      aria-live="polite"
    43→      aria-atomic="true"
    44→      className="absolute w-px h-px overflow-hidden whitespace-nowrap"
    45→      style={{ clip: 'rect(0 0 0 0)' }}
    46→    />
    47→  )
    48→}
    49→
    50→/**
    51→ * Announce a message to screen readers via the live region.
    52→ * Debounced at 60ms so rapid calls don't flood the SR queue.
    53→ */
    54→export function announce(message: string) {
    55→  if (typeof window === 'undefined') return
    56→  const el = document.getElementById('ve-live-region')
    57→  if (!el) return
    58→
    59→  if (_announceTimer) clearTimeout(_announceTimer)
    60→  _announceTimer = setTimeout(() => {
    61→    el.textContent = ''
    62→    requestAnimationFrame(() => {
    63→      el.textContent = message
    64→    })
    65→  }, 60)
    66→}

   327→  value: T

   331→
   332→export function SegmentedControl<T extends string>({
   333→  label,
   334→  options,
   335→  value,
   336→  onChange,
   337→  className,
   338→}: SegmentedControlProps<T>) {
   339→  const baseId = useId()

   357→        style={{
   358→          left: activeIndex >= 0 ? `calc(${activeIndex} * (100% - 4px) / ${options.length} + 2px)` : 0,
   359→          width: `calc((100% - 4px) / ${options.length})`,

   372→            <input
   373→              id={id}
   374→              type="radio"
   375→              name={baseId}
   376→              value={opt.value}
   377→              checked={active}
   378→              onChange={() => onChange(opt.value)}

   380→              tabIndex={-1}
   381→            />
   382→            <label
   383→              htmlFor={id}

   387→                  e.preventDefault()
   388→                  onChange(opt.value)

   390→              }}
   391→              title={opt.label}
   392→              className={cn(

   407→            </label>

   579→// 9. ColorField — Color picker with swatch + hex + alpha
   580→// ═══════════════════════════════════════════════════════════════════════════
   581→
   582→export interface ColorFieldProps {
   583→  label: string
   584→  value: string
   585→  onChange: (v: string) => void
   586→  /** Enable alpha channel */
   587→  alpha?: boolean
   588→  className?: string
   589→}
   590→
   591→export function ColorField({
   592→  label,
   593→  value,
   594→  onChange,
   595→  alpha = false,
   596→  className,
   597→}: ColorFieldProps) {
   598→  // Extract hex and alpha from value
   599→  const hexPart = value.startsWith('#') ? value.slice(0, 7) : '#000000'
   600→  const alphaPart = value.length === 9 ? parseInt(value.slice(7, 9), 16) / 255 : 1
   601→
   602→  const handleHexChange = useCallback(
   603→    (newHex: string) => {
   604→      if (/^#[0-9a-fA-F]{0,6}$/.test(newHex)) {
   605→        if (newHex.length === 7 && alpha) {
   606→          const aHex = Math.round(alphaPart * 255).toString(16).padStart(2, '0')
   607→          onChange(newHex + aHex)
   608→        } else {
   609→          onChange(newHex)
   610→        }
   611→      }
   612→    },
   613→    [alpha, alphaPart, onChange]
   614→  )
   615→
   616→  const handleAlphaChange = useCallback(
   617→    (newAlpha: number) => {
   618→      const aHex = Math.round(newAlpha * 255).toString(16).padStart(2, '0')
   619→      onChange(hexPart + aHex)
   620→    },
   621→    [hexPart, onChange]
   622→  )
   623→
   624→  return (
   625→    <div className={cn('flex flex-col gap-1.5', className)}>
   626→      <span className="text-[13px] font-medium text-neutral-800 dark:text-neutral-200">
   627→        {label}
   628→      </span>
   629→      <div className="flex items-center gap-2">
   630→        {/* Swatch */}
   631→        <div
   632→          className="relative w-10 h-10 rounded-lg border border-neutral-300 dark:border-neutral-600 overflow-hidden shrink-0"
   633→          style={{ background: hexPart }}
   634→        >
   635→          {/* Alpha checkerboard */}
   636→          {alpha && (
   637→            <div
   638→              className="absolute inset-0"
   639→              style={{
   640→                background: `linear-gradient(45deg, #ccc 25%, transparent 25%, transparent 75%, #ccc 75%), linear-gradient(45deg, #ccc 25%, transparent 25%, transparent 75%, #ccc 75%)`,
   641→                backgroundSize: '8px 8px',
   642→                backgroundPosition: '0 0, 4px 4px',
   643→                opacity: 1 - alphaPart,
   644→              }}
   645→            />
   646→          )}
   647→          <input
   648→            type="color"
   649→            aria-label={`${label} color`}
   650→            value={hexPart}
   651→            onChange={(e) => handleHexChange(e.target.value)}
   652→            className="absolute -inset-1 w-12 h-12 border-none cursor-pointer opacity-0"
   653→          />
   654→        </div>
   655→        {/* Hex input */}
   656→        <input
   657→          type="text"
   658→          aria-label={`${label} hex`}
   659→          value={hexPart.toUpperCase()}
   660→          onChange={(e) => handleHexChange(e.target.value)}
   661→          className={cn(
   662→            'flex-1 min-h-[44px] px-2.5',
   663→            'text-[13px] font-mono',
   664→            'text-neutral-800 dark:text-neutral-200',
   665→            'bg-white dark:bg-neutral-800',
   666→            'border border-neutral-300 dark:border-neutral-600',
   667→            'rounded-lg',
   668→            'outline-none',
   669→            'transition-colors duration-150',
   670→            'focus-visible:ring-2 focus-visible:ring-blue-500',
   671→          )}
   672→        />
   673→      </div>
   674→      {/* Alpha slider */}
   675→      {alpha && (
   676→        <div className="flex items-center gap-2 mt-1">
   677→          <span className="text-[11px] text-neutral-500 dark:text-neutral-400 w-8">α</span>
   678→          <input
   679→            type="range"
   680→            aria-label={`${label} alpha`}
   681→            min={0}
   682→            max={1}
   683→            step={0.01}
   684→            value={alphaPart}
   685→            onChange={(e) => handleAlphaChange(Number(e.target.value))}
   686→            className="flex-1 h-1.5 rounded-full accent-blue-600 cursor-pointer"
   687→          />
   688→          <span className="text-[11px] tabular-nums text-neutral-500 dark:text-neutral-400 w-8 text-right">
   689→            {Math.round(alphaPart * 100)}%
   690→          </span>
   691→        </div>
   692→      )}
   693→    </div>
   694→  )
   695→}
