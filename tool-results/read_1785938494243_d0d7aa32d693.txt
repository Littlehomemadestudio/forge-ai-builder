   193→// ─── Resolve any style/palette ID into a usable preset ──────────────────────
   194→// The builderStyle can be any palette ID (e.g. 'saffron', 'persian-carpet')
   195→// which is NOT in STYLE_PRESETS. This function resolves it by:
   196→// 1. Checking STYLE_PRESETS first (for the 8 core styles)
   197→// 2. Looking up ALL_PALETTES and constructing a synthetic preset from palette data
   198→// 3. Falling back to the 'dark' preset if neither found
   199→
   200→type ResolvedStylePreset = {
   201→  label: string;
   202→  colors: { bg: string; surface: string; text: string; muted: string; accent: string; border: string };
   203→  fontStack: string;
   204→  mood: string;
   205→  paletteId?: string; // track the palette ID for advanced options color override
   206→};
   207→
   208→function resolveStylePreset(styleId: string): ResolvedStylePreset {
   209→  // 1. Direct match in STYLE_PRESETS
   210→  if (STYLE_PRESETS[styleId]) {
   211→    return { ...STYLE_PRESETS[styleId], paletteId: styleId };
   212→  }
   213→
   214→  // 2. Look up in ALL_PALETTES
   215→  const palette = ALL_PALETTES.find(p => p.id === styleId);
   216→  if (palette) {
   217→    // Determine a border color from the palette (derive if not explicit)
   218→    const border = palette.colors.muted || '#E2E8F0';
   219→    const fontStack = palette.fontSuggestion
   220→      ? `'${palette.fontSuggestion}', 'Inter', -apple-system, BlinkMacSystemFont, sans-serif`
   221→      : "'Inter', -apple-system, BlinkMacSystemFont, sans-serif";
   222→
   223→    return {
   224→      label: `${palette.name} — ${palette.mood}`,
   225→      colors: {
   226→        bg: palette.colors.background,
   227→        surface: palette.colors.surface,
   228→        text: palette.colors.text,
   229→        muted: palette.colors.muted,
   230→        accent: palette.colors.primary,
   231→        border,
   232→      },
   233→      fontStack,
   234→      mood: palette.mood,
   235→      paletteId: styleId,
   236→    };
   237→  }
   238→
   239→  // 3. Fallback to dark
   240→  return { ...STYLE_PRESETS.dark, paletteId: styleId };
   241→}

   272→function extractHtmlFromResponse(content: string): string {
   273→  if (!content) return '';
   274→  const codeBlockMatch = content.match(/```(?:html)?\s*\n([\s\S]*?)```/i);
   275→  if (codeBlockMatch) {
   276→    return codeBlockMatch[1].trim();
   277→  }
   278→  const htmlStart = content.toLowerCase().indexOf('<!doctype');
   279→  if (htmlStart !== -1) {
   280→    const htmlEnd = content.toLowerCase().lastIndexOf('</html>');
   281→    if (htmlEnd !== -1) {
   282→      return content.substring(htmlStart, htmlEnd + '</html>'.length).trim();
   283→    }
   284→  }
   285→  const htmlTagStart = content.toLowerCase().indexOf('<html');
   286→  if (htmlTagStart !== -1) {
   287→    const htmlEnd = content.toLowerCase().lastIndexOf('</html>');
   288→    if (htmlEnd !== -1) {
   289→      return content.substring(htmlTagStart, htmlEnd + '</html>'.length).trim();
   290→    }
   291→  }
   292→  return content.trim();
   293→}

   295→function extractCssFromHtml(html: string): string {
   296→  const match = html.match(/<style[^>]*>([\s\S]*?)<\/style>/i);
   297→  return match ? match[1].trim() : '';

   299→
   300→function extractJsFromHtml(html: string): string {
   301→  const matches = html.match(/<script[^>]*>([\s\S]*?)<\/script>/gi);
   302→  if (!matches) return '';
   303→  let js = '';
   304→  for (const tag of matches) {
   305→    if (tag.match(/<script[^>]*\bsrc=/i)) continue;
   306→    const contentMatch = tag.match(/<script[^>]*>([\s\S]*?)<\/script>/i);
   307→    if (contentMatch) js += contentMatch[1] + '\n';
   308→  }
   309→  return js.trim();
   310→}

   312→// ─── Phase 5: Animation Library Injection ─────────────────────────────────
   313→// Maps the parser's animation IDs (legacy names) to the new library IDs.
   314→const LEGACY_ANIM_MAP: Record<string, AnimationId> = {
   315→  'glassmorphism': 'glassmorphism',
   316→  'parallax': 'parallax',
   317→  'skeleton-loader': 'skeleton',
   318→  '360-rotate': 'rotate-360',
   319→  'count-up': 'count-up',
   320→  'shake-animation': 'shake-cart',
   321→};

   323→/**
   324→ * Build the set of animation IDs that should be injected for this prompt.
   325→ * Combines:
   326→ *   1. Animations detected by the new library's keyword matcher
   327→ *   2. Animations already detected by the legacy parser
   328→ *   3. Glassmorphism if the parser flagged it as a theme keyword
   329→ */
   330→function resolveAnimationIds(parsed: ParsedPrompt): AnimationId[] {
   331→  const ids = new Set<AnimationId>();
   332→
   333→  // From the new library (keyword detection against the raw prompt text)
   334→  for (const id of detectAnimations(parsed._rawPrompt || '')) {
   335→    ids.add(id);
   336→  }
   337→
   338→  // From the legacy parser's animation list
   339→  for (const legacy of parsed.animations) {
   340→    const mapped = LEGACY_ANIM_MAP[legacy];
   341→    if (mapped) ids.add(mapped);
   342→  }
   343→
   344→  // Glassmorphism is sometimes flagged as a theme keyword rather than an animation
   345→  if (parsed.themeKeywords.includes('glassmorphism')) {
   346→    ids.add('glassmorphism');
   347→  }
   348→
   349→  return Array.from(ids);
   350→}

   352→/**
   353→ * Inject the guaranteed-working animation CSS/JS bundle into the AI-generated
   354→ * HTML. CSS is appended inside the first <style> tag (or a new one is created).
   355→ * JS is inserted right before </body>.
   356→ *
   357→ * The injected CSS uses scoped class names (.forge-glass, .forge-parallax-bg,
   358→ * .forge-skeleton, .forge-rotate-360, [data-count-target], .forge-shake) so it
   359→ * never clobbers the AI's own styles. The injected JS self-initializes on
   360→ * DOMContentLoaded and is safe to include even if no matching elements exist.
   361→ */
   362→function injectAnimationBundle(html: string, ids: AnimationId[]): string {
   363→  if (ids.length === 0) return html;
   364→  const bundle = buildAnimationBundle(ids);
   365→  if (!bundle.css && !bundle.js) return html;
   366→
   367→  let out = html;
   368→
   369→  // ── Inject CSS ──
   370→  const cssBlock = `\n/* === FORGE ANIMATION LIBRARY (auto-injected) === */\n/* Includes: ${bundle.namesEn.join(', ')} */\n${bundle.css}\n/* === END FORGE ANIMATION LIBRARY === */\n`;
   371→
   372→  const styleMatch = out.match(/<style[^>]*>/i);
   373→  if (styleMatch && styleMatch.index !== undefined) {
   374→    // Insert right after the opening <style> tag
   375→    const insertAt = styleMatch.index + styleMatch[0].length;
   376→    out = out.slice(0, insertAt) + cssBlock + out.slice(insertAt);
   377→  } else {
   378→    // No <style> tag — inject one in the <head>
   379→    const headMatch = out.match(/<head[^>]*>/i);
   380→    if (headMatch && headMatch.index !== undefined) {
   381→      const insertAt = headMatch.index + headMatch[0].length;
   382→      out = out.slice(0, insertAt) + `\n<style id="forge-animations">${cssBlock}</style>\n` + out.slice(insertAt);
   383→    }
   384→  }
   385→
   386→  // ── Inject JS ──
   387→  if (bundle.js) {
   388→    const jsBlock = `\n<script id="forge-animations-js">\n/* === FORGE ANIMATION LIBRARY (auto-injected) === */\n/* Includes: ${bundle.namesEn.join(', ')} */\n${bundle.js}\n/* === END FORGE ANIMATION LIBRARY === */\n</script>\n`;
   389→    const bodyEnd = out.toLowerCase().lastIndexOf('</body>');
   390→    if (bodyEnd !== -1) {
   391→      out = out.slice(0, bodyEnd) + jsBlock + out.slice(bodyEnd);
   392→    } else {
   393→      out = out + jsBlock;
   394→    }
   395→  }
   396→
   397→  return out;
   398→}

   400→function extractSiteNameFromPrompt(prompt: string, industry: Industry): string {
   401→  const patterns = [
   402→    /(?:called|named)\s+["']?([A-Z][\w&'-]+(?:\s+[A-Z][\w&'-]+)?)/,
   403→    /for\s+(?:a|an|the)\s+(?:coffee shop|cafe|restaurant|studio|agency|brand|company|startup|business|shop|store)\s+(?:called|named)\s+["']?([A-Z][\w&'-]+)/,
   404→  ];
   405→  for (const p of patterns) {
   406→    const m = prompt.match(p);
   407→    if (m && m[1]) return m[1].trim();
   408→  }
   409→  const quoted = prompt.match(/["']([A-Z][\w\s&'-]{2,30})["']/);
   410→  if (quoted && quoted[1]) return quoted[1].trim();
   411→  return INDUSTRY_META[industry].defaultSiteName;
   412→}

   569→function hexToRgb(hex: string): [number, number, number] {
   570→  const h = hex.replace('#', '');
   571→  return [
   572→    parseInt(h.substring(0, 2), 16),
   573→    parseInt(h.substring(2, 4), 16),
   574→    parseInt(h.substring(4, 6), 16),
   575→  ];
   576→}
   577→
   578→function lightenColor(hex: string, percent: number): string {
   579→  const [r, g, b] = hexToRgb(hex);
   580→  const factor = percent / 100;
   581→  const nr = Math.min(255, Math.round(r + (255 - r) * factor));
   582→  const ng = Math.min(255, Math.round(g + (255 - g) * factor));
   583→  const nb = Math.min(255, Math.round(b + (255 - b) * factor));
   584→  return '#' + [nr, ng, nb].map(v => v.toString(16).padStart(2, '0')).join('').toUpperCase();
   585→}
   586→
   587→function isLightColor(hex: string): boolean {
   588→  const [r, g, b] = hexToRgb(hex);
   589→  // Perceived luminance
   590→  const lum = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   591→  return lum > 0.5;
   592→}

  1195→function darkenColor(hex: string, percent: number): string {
  1196→  const [r, g, b] = hexToRgb(hex);
  1197→  const factor = percent / 100;
  1198→  const nr = Math.max(0, Math.round(r * (1 - factor)));
  1199→  const ng = Math.max(0, Math.round(g * (1 - factor)));
  1200→  const nb = Math.max(0, Math.round(b * (1 - factor)));
  1201→  return '#' + [nr, ng, nb].map(v => v.toString(16).padStart(2, '0')).join('').toUpperCase();
  1202→}

  1782→function sleep(ms: number) {

  1786→/**
  1787→ * Stream the AI completion for a page, retrying transient errors (429 / 5xx /
  1788→ * timeout) but only BEFORE the first token is yielded — once streaming has
  1789→ * started we cannot safely retry without duplicating output.
  1790→ */
  1791→async function streamPageContent(
  1792→  systemPrompt: string,
  1793→  userPrompt: string,
  1794→  onDelta: (chunk: string) => void
  1795→): Promise<string> {
  1796→  let lastError: any = null;
  1797→
  1798→  for (let attempt = 0; attempt < MAX_RETRIES; attempt++) {
  1799→    let yieldedAny = false;
  1800→    try {
  1801→      let acc = '';
  1802→      for await (const chunk of streamChatDeltas(systemPrompt, userPrompt, {
  1803→        maxTokens: AI_MAX_TOKENS,
  1804→        timeoutMs: AI_REQUEST_TIMEOUT_MS,
  1805→      })) {
  1806→        yieldedAny = true;
  1807→        acc += chunk;
  1808→        onDelta(chunk);
  1809→      }
  1810→      if (acc.length < 200) {
  1811→        throw new Error(`AI returned empty or too-short content (len=${acc.length})`);
  1812→      }
  1813→      return acc;
  1814→    } catch (err: any) {
  1815→      lastError = err;
  1816→      const msg = String(err?.message || err);
  1817→      const is429 = msg.includes('429') || msg.includes('Too many requests') || /rate/i.test(msg);
  1818→      const is5xx = ['502', '503', '504', '500'].some(c => msg.includes(c));
  1819→      const isTimeout =
  1820→        msg.includes('timeout') || msg.includes('aborted') || msg.includes('ECONNRESET') || msg.includes('AbortError');
  1821→      if (attempt < MAX_RETRIES - 1 && !yieldedAny && (is429 || is5xx || isTimeout)) {
  1822→        const wait = 3000 * Math.pow(2, attempt);
  1823→        console.log(`[generate] attempt ${attempt + 1} failed (${msg.slice(0, 120)}), retrying in ${wait}ms`);
  1824→        await sleep(wait);
  1825→        continue;
  1826→      }
  1827→      throw err;
  1828→    }
  1829→  }
  1830→  throw lastError || new Error('AI generation failed after retries');
  1831→}
