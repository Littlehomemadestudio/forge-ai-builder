    43→    43→      aria-atomic="true"

    50→    50→/**
    51→    51→ * Announce a message to screen readers via the live region.
    52→    52→ * Debounced at 60ms so rapid calls don't flood the SR queue.
    53→    53→ */
    54→    54→export function announce(message: string) {
    55→    55→  if (typeof window === 'undefined') return
    56→    56→  const el = document.getElementById('ve-live-region')
    57→    57→  if (!el) return
    58→    58→
    59→    59→  if (_announceTimer) clearTimeout(_announceTimer)
    60→    60→  _announceTimer = setTimeout(() => {
    61→    61→    el.textContent = ''
    62→    62→    requestAnimationFrame(() => {
    63→    63→      el.textContent = message
    64→    64→    })
    65→    65→  }, 60)
    66→    66→}

   321→   321→// 6. SegmentedControl — Toggle group with active indicator
   322→   322→// ═══════════════════════════════════════════════════════════════════════════
   323→   323→
   324→   324→export interface SegmentedControlProps<T extends string> {
   325→   325→  label: string
   326→   326→  options: { value: T; label: string; icon?: React.ReactNode }[]
   327→   327→  value: T
   328→   328→  onChange: (v: T) => void
   329→   329→  className?: string

   331→   331→
   332→   332→export function SegmentedControl<T extends string>({
   333→   333→  label,
   334→   334→  options,
   335→   335→  value,
   336→   336→  onChange,
   337→   337→  className,
   338→   338→}: SegmentedControlProps<T>) {

   340→   340→  const activeIndex = options.findIndex((o) => o.value === value)
   341→   341→
   342→   342→  return (
   343→   343→    <div
   344→   344→      role="radiogroup"
   345→   345→      aria-label={label}
   346→   346→      className={cn(
   347→   347→        'relative inline-flex gap-0.5 p-0.5',
   348→   348→        'bg-neutral-100 dark:bg-neutral-800',
   349→   349→        'rounded-lg',
   350→   350→        'border border-neutral-200 dark:border-neutral-700',

   356→   356→        className="absolute top-0.5 bottom-0.5 rounded-md bg-white dark:bg-neutral-700 shadow-sm"
   357→   357→        style={{
   358→   358→          left: activeIndex >= 0 ? `calc(${activeIndex} * (100% - 4px) / ${options.length} + 2px)` : 0,
   359→   359→          width: `calc((100% - 4px) / ${options.length})`,
   360→   360→        }}
   361→   361→        animate={{
   362→   362→          x: activeIndex * 100,

   367→   367→      {options.map((opt, i) => {

   376→   376→              value={opt.value}

   378→   378→              onChange={() => onChange(opt.value)}
   379→   379→              className="sr-only"
   380→   380→              tabIndex={-1}
   381→   381→            />
   382→   382→            <label
   383→   383→              htmlFor={id}
   384→   384→              tabIndex={active ? 0 : -1}

   387→   387→                  e.preventDefault()
   388→   388→                  onChange(opt.value)
   389→   389→                }
   390→   390→              }}
   391→   391→              title={opt.label}
   392→   392→              className={cn(
   393→   393→                'relative z-10',
   394→   394→                'min-h-[32px] min-w-[40px] px-2.5',
   395→   395→                'flex items-center justify-center gap-1.5',
   396→   396→                'rounded-md',
   397→   397→                'text-[13px] font-medium',
   398→   398→                'cursor-pointer select-none',
   399→   399→                'transition-colors duration-150',
   400→   400→                active
   401→   401→                  ? 'text-neutral-900 dark:text-neutral-100'
   402→   402→                  : 'text-neutral-500 dark:text-neutral-400 hover:text-neutral-700 dark:hover:text-neutral-300',
   403→   403→              )}
   404→   404→            >
   405→   405→              {opt.icon && <span className="inline-flex">{opt.icon}</span>}
   406→   406→              {opt.label}
   407→   407→            </label>

   409→   409→        )
   410→   410→      })}
   411→   411→    </div>
   412→   412→  )
   413→   413→}

   579→   579→// 9. ColorField — Color picker with swatch + hex + alpha
   580→   580→// ═══════════════════════════════════════════════════════════════════════════
   581→   581→
   582→   582→export interface ColorFieldProps {
   583→   583→  label: string
   584→   584→  value: string
   585→   585→  onChange: (v: string) => void
   586→   586→  /** Enable alpha channel */
   587→   587→  alpha?: boolean
   588→   588→  className?: string
   589→   589→}
   590→   590→
   591→   591→export function ColorField({
   592→   592→  label,
   593→   593→  value,
   594→   594→  onChange,
   595→   595→  alpha = false,
   596→   596→  className,
   597→   597→}: ColorFieldProps) {
   598→   598→  // Extract hex and alpha from value
   599→   599→  const hexPart = value.startsWith('#') ? value.slice(0, 7) : '#000000'
   600→   600→  const alphaPart = value.length === 9 ? parseInt(value.slice(7, 9), 16) / 255 : 1
   601→   601→
   602→   602→  const handleHexChange = useCallback(
   603→   603→    (newHex: string) => {
   604→   604→      if (/^#[0-9a-fA-F]{0,6}$/.test(newHex)) {
   605→   605→        if (newHex.length === 7 && alpha) {
   606→   606→          const aHex = Math.round(alphaPart * 255).toString(16).padStart(2, '0')
   607→   607→          onChange(newHex + aHex)
   608→   608→        } else {
   609→   609→          onChange(newHex)
   610→   610→        }
   611→   611→      }
   612→   612→    },
   613→   613→    [alpha, alphaPart, onChange]
   614→   614→  )
   615→   615→
   616→   616→  const handleAlphaChange = useCallback(
   617→   617→    (newAlpha: number) => {
   618→   618→      const aHex = Math.round(newAlpha * 255).toString(16).padStart(2, '0')
   619→   619→      onChange(hexPart + aHex)
   620→   620→    },
   621→   621→    [hexPart, onChange]
   622→   622→  )
   623→   623→
   624→   624→  return (
   625→   625→    <div className={cn('flex flex-col gap-1.5', className)}>
   626→   626→      <span className="text-[13px] font-medium text-neutral-800 dark:text-neutral-200">
   627→   627→        {label}
   628→   628→      </span>
   629→   629→      <div className="flex items-center gap-2">
   630→   630→        {/* Swatch */}
   631→   631→        <div
   632→   632→          className="relative w-10 h-10 rounded-lg border border-neutral-300 dark:border-neutral-600 overflow-hidden shrink-0"
   633→   633→          style={{ background: hexPart }}
   634→   634→        >
   635→   635→          {/* Alpha checkerboard */}
   636→   636→          {alpha && (
   637→   637→            <div
   638→   638→              className="absolute inset-0"
   639→   639→              style={{
   640→   640→                background: `linear-gradient(45deg, #ccc 25%, transparent 25%, transparent 75%, #ccc 75%), linear-gradient(45deg, #ccc 25%, transparent 25%, transparent 75%, #ccc 75%)`,
   641→   641→                backgroundSize: '8px 8px',
   642→   642→                backgroundPosition: '0 0, 4px 4px',
   643→   643→                opacity: 1 - alphaPart,
   644→   644→              }}
   645→   645→            />
   646→   646→          )}
   647→   647→          <input
   648→   648→            type="color"
   649→   649→            aria-label={`${label} color`}
   650→   650→            value={hexPart}
   651→   651→            onChange={(e) => handleHexChange(e.target.value)}
   652→   652→            className="absolute -inset-1 w-12 h-12 border-none cursor-pointer opacity-0"
   653→   653→          />
   654→   654→        </div>
   655→   655→        {/* Hex input */}
   656→   656→        <input
   657→   657→          type="text"
   658→   658→          aria-label={`${label} hex`}
   659→   659→          value={hexPart.toUpperCase()}
   660→   660→          onChange={(e) => handleHexChange(e.target.value)}
   661→   661→          className={cn(
   662→   662→            'flex-1 min-h-[44px] px-2.5',
   663→   663→            'text-[13px] font-mono',
   664→   664→            'text-neutral-800 dark:text-neutral-200',
   665→   665→            'bg-white dark:bg-neutral-800',
   666→   666→            'border border-neutral-300 dark:border-neutral-600',
   667→   667→            'rounded-lg',
   668→   668→            'outline-none',
   669→   669→            'transition-colors duration-150',
   670→   670→            'focus-visible:ring-2 focus-visible:ring-blue-500',
   671→   671→          )}
   672→   672→        />
   673→   673→      </div>
   674→   674→      {/* Alpha slider */}
   675→   675→      {alpha && (
   676→   676→        <div className="flex items-center gap-2 mt-1">
   677→   677→          <span className="text-[11px] text-neutral-500 dark:text-neutral-400 w-8">α</span>
   678→   678→          <input
   679→   679→            type="range"
   680→   680→            aria-label={`${label} alpha`}
   681→   681→            min={0}
   682→   682→            max={1}
   683→   683→            step={0.01}
   684→   684→            value={alphaPart}
   685→   685→            onChange={(e) => handleAlphaChange(Number(e.target.value))}
   686→   686→            className="flex-1 h-1.5 rounded-full accent-blue-600 cursor-pointer"
   687→   687→          />
   688→   688→          <span className="text-[11px] tabular-nums text-neutral-500 dark:text-neutral-400 w-8 text-right">
   689→   689→            {Math.round(alphaPart * 100)}%
   690→   690→          </span>
   691→   691→        </div>
   692→   692→      )}
   693→   693→    </div>
   694→   694→  )
   695→   695→}
