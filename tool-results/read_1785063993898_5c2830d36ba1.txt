  1407→interface PageGenState {
  1408→  id: string
  1409→  name: string
  1410→  status: 'pending' | 'generating' | 'done' | 'error'
  1411→  error?: string
  1412→  html?: string

  1414→  js?: string

  1417→
  1418→function GeneratingPhase() {
  1419→  const {
  1420→    builderPrompt, builderIndustry, builderStyle,
  1421→    builderAdvancedOptions,
  1422→    setGeneratedPages, setGeneratedSiteName,
  1423→    setCurrentPreviewPage, setIsGenerating, setBuilderPhase,
  1424→    setGenerationProgress, setGenerationStatus,
  1425→    generationStatus,
  1426→  } = useAppStore()
  1427→
  1428→  // Determine which core pages to generate based on pageConfigs
  1429→  const pagesToGenerate = CORE_PAGE_ORDER.filter(cp =>
  1430→    builderAdvancedOptions.pageConfigs.find(pc => pc.id === cp.id && pc.enabled)
  1431→  )
  1432→
  1433→  const [pageStates, setPageStates] = useState<PageGenState[]>(
  1434→    pagesToGenerate.map(p => ({ id: p.id, name: p.name, status: 'pending' }))
  1435→  )
  1436→  const [elapsedSeconds, setElapsedSeconds] = useState(0)
  1437→  const [globalError, setGlobalError] = useState<string | null>(null)
  1438→  const cancelRef = useRef(false)
  1439→
  1440→  // Tick elapsed timer
  1441→  useEffect(() => {
  1442→    const interval = setInterval(() => {
  1443→      if (!cancelRef.current) setElapsedSeconds(s => s + 1)
  1444→    }, 1000)
  1445→    return () => clearInterval(interval)
  1446→  }, [])
  1447→
  1448→  // Run page generations SEQUENTIALLY
  1449→  useEffect(() => {
  1450→    cancelRef.current = false
  1451→    let cancelled = false
  1452→
  1453→    async function generateAll() {
  1454→      const collected: { id: string; name: string; route: string; html: string; css: string; js?: string }[] = []
  1455→      let siteName = ''
  1456→
  1457→      for (let i = 0; i < pagesToGenerate.length; i++) {
  1458→        if (cancelled || cancelRef.current) return
  1459→        const pageInfo = pagesToGenerate[i]
  1460→
  1461→        setPageStates(prev => prev.map(p => p.id === pageInfo.id ? { ...p, status: 'generating' } : p))
  1462→        setGenerationStatus(`Generating ${pageInfo.name} page…`)
  1463→        setGenerationProgress(Math.round((i / pagesToGenerate.length) * 100))
  1464→
  1465→        try {
  1466→          // Build request body with advanced options
  1467→          const requestBody = {
  1468→            prompt: builderPrompt,
  1469→            industry: builderIndustry,
  1470→            style: builderStyle,
  1471→            page: pageInfo.id,
  1472→            siteName: builderAdvancedOptions.brandName || undefined,
  1473→            // Advanced options for the server to use in prompt construction
  1474→            advancedOptions: {
  1475→              complexity: builderAdvancedOptions.complexity,
  1476→              pageLength: builderAdvancedOptions.pageConfigs.find(p => p.id === pageInfo.id)?.length || builderAdvancedOptions.pageLength,
  1477→              layoutDensity: builderAdvancedOptions.layoutDensity,
  1478→              animationLevel: builderAdvancedOptions.animationLevel,
  1479→              responsivePriority: builderAdvancedOptions.responsivePriority,
  1480→              contentTone: builderAdvancedOptions.contentTone,
  1481→              navigationStyle: builderAdvancedOptions.navigationStyle,
  1482→              seoLevel: builderAdvancedOptions.seoLevel,
  1483→              accessibilityLevel: builderAdvancedOptions.accessibilityLevel,
  1484→              imageStyle: builderAdvancedOptions.imageStyle,
  1485→              ctaStyle: builderAdvancedOptions.ctaStyle,
  1486→              fontFamily: builderAdvancedOptions.fontFamily,
  1487→              colorScheme: builderAdvancedOptions.colorScheme,

  1491→              includeFeatures: builderAdvancedOptions.includeFeatures,
  1492→              includeTestimonials: builderAdvancedOptions.includeTestimonials,

  1495→              includeNewsletter: builderAdvancedOptions.includeNewsletter,
  1496→              includeCTA: builderAdvancedOptions.includeCTA,

  1499→              includeSocialLinks: builderAdvancedOptions.includeSocialLinks,
  1500→              includeContactForm: builderAdvancedOptions.includeContactForm,

  1506→          const startRes = await fetch('/api/generate', {
  1507→            method: 'POST',
  1508→            headers: { 'Content-Type': 'application/json' },

  1514→          if (!startRes.ok) {

  1518→
  1519→          const startData = await startRes.json()
  1520→          const jobId = startData?.jobId
  1521→          if (!jobId) {
  1522→            throw new Error('Server did not return a jobId')
  1523→          }
  1524→
  1525→          // Poll until done/error
  1526→          let data: any = null
  1527→          let streamError: string | null = null
  1528→          const POLL_INTERVAL_MS = 2500
  1529→          const MAX_POLL_DURATION_MS = 8 * 60 * 1000
  1530→          const pollStart = Date.now()
  1531→
  1532→          while (true) {
  1533→            if (cancelled || cancelRef.current) return
  1534→
  1535→            const elapsed = Date.now() - pollStart
  1536→            if (elapsed > MAX_POLL_DURATION_MS) {
  1537→              throw new Error(`Timed out after ${Math.round(elapsed / 1000)}s waiting for ${pageInfo.name} page`)
  1538→            }
  1539→
  1540→            if (elapsed > 0) {
  1541→              await new Promise<void>(r => setTimeout(r, POLL_INTERVAL_MS))
  1542→              if (cancelled || cancelRef.current) return
  1543→            }
  1544→
  1545→            const statusRes = await fetch(`/api/generate?jobId=${encodeURIComponent(jobId)}`, {
  1546→              method: 'GET',
  1547→              headers: { 'Cache-Control': 'no-cache' },
  1548→            })
  1549→
  1550→            if (!statusRes.ok) {
  1551→              if (statusRes.status === 404) {
  1552→                throw new Error('Generation job disappeared — please retry')
  1553→              }
  1554→              const errBody = await statusRes.json().catch(() => ({ error: `HTTP ${statusRes.status}` }))
  1555→              throw new Error(errBody.error || `Status poll HTTP ${statusRes.status}`)
  1556→            }
  1557→
  1558→            const status = await statusRes.json()
  1559→            if (cancelled || cancelRef.current) return

  1563→                `Generating ${pageInfo.name} page… (${status.heartbeats} AI ticks, ${Math.round((Date.now() - pollStart) / 1000)}s)`

  1565→            }
  1566→
  1567→            if (status.status === 'done') {
  1568→              data = status
  1569→              break
  1570→            }
  1571→            if (status.status === 'error') {
  1572→              streamError = status.error || 'Generation failed'
  1573→              break
  1574→            }
  1575→          }
  1576→
  1577→          if (cancelled || cancelRef.current) return
  1578→
  1579→          if (streamError) {
  1580→            throw new Error(streamError)
  1581→          }
  1582→          if (!data?.result?.page?.html) {
  1583→            throw new Error('AI returned no HTML')
  1584→          }
  1585→
  1586→          if (data.result.siteName && !siteName) siteName = data.result.siteName
  1587→
  1588→          const completed: PageGenState = {
  1589→            id: pageInfo.id,
  1590→            name: data.result.page.name || pageInfo.name,
  1591→            status: 'done',
  1592→            html: data.result.page.html,
  1593→            css: data.result.page.css || '',
  1594→            js: data.result.page.js || '',
  1595→          }
  1596→
  1597→          collected.push({
  1598→            id: `page-${pageInfo.id}`,
  1599→            name: completed.name,
  1600→            route: pageInfo.id === 'home' ? '/' : `/${pageInfo.id}`,
  1601→            html: completed.html || '',
  1602→            css: completed.css || '',
  1603→            js: completed.js,
  1604→          })
  1605→
  1606→          setPageStates(prev => prev.map(p => p.id === pageInfo.id ? completed : p))
  1607→          setGenerationProgress(Math.round(((i + 1) / pagesToGenerate.length) * 100))
  1608→        } catch (err: any) {
  1609→          if (cancelled || cancelRef.current) return
  1610→          const msg = err?.message || 'Generation failed'
  1611→          setPageStates(prev => prev.map(p => p.id === pageInfo.id ? { ...p, status: 'error', error: msg } : p))
  1612→          setGlobalError(msg)
  1613→          return
  1614→        }
  1615→      }
  1616→
  1617→      if (cancelled || cancelRef.current) return
  1618→
  1619→      setGeneratedPages(collected)
  1620→      setGeneratedSiteName(siteName || builderAdvancedOptions.brandName || 'Untitled Site')
  1621→      setCurrentPreviewPage(collected[0].id)
  1622→      setGenerationProgress(100)
  1623→      setGenerationStatus('Done!')
  1624→      setIsGenerating(false)
  1625→      setBuilderPhase('preview')
  1626→    }
  1627→
  1628→    generateAll()
  1629→
  1630→    return () => {
  1631→      cancelled = true
  1632→      cancelRef.current = true
  1633→    }
  1634→  }, [])
  1635→
  1636→  const handleCancel = () => {
  1637→    cancelRef.current = true
  1638→    setIsGenerating(false)
  1639→    setGenerationProgress(0)
  1640→    setGenerationStatus('')
