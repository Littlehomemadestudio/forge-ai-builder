   303→  window.addEventListener('message', function(e) {
   304→    const msg = e.data;
   305→    if (!msg || !msg.type) return;
   306→
   307→    switch (msg.type) {
   308→      case 'apply-style': {
   309→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   310→        if (el) el.style.setProperty(msg.data.property, msg.data.value, 'important');
   311→        break;
   312→      }
   313→      case 'apply-content': {
   314→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   315→        if (el) {
   316→          el.innerHTML = msg.data.content;
   317→          assignIds();
   318→          if (selectedId === msg.data.elementId) {
   319→            showSelection(selectedId);
   320→            sendElementInfo(el);
   321→          }
   322→        }
   323→        break;
   324→      }
   325→      case 'move-element': {
   326→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   327→        if (el && el.parentElement) {
   328→          const dir = msg.data.direction;
   329→          if (dir === 'up' && el.previousElementSibling) {
   330→            el.parentElement.insertBefore(el, el.previousElementSibling);
   331→          } else if (dir === 'down' && el.nextElementSibling) {
   332→            el.parentElement.insertBefore(el.nextElementSibling, el);
   333→          }
   334→          assignIds();
   335→          if (selectedId === msg.data.elementId) {
   336→            showSelection(selectedId);
   337→            sendElementInfo(el);
   338→          }
   339→        }
   340→        break;
   341→      }
   342→      case 'duplicate-element': {
   343→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   344→        if (el && el.parentElement) {
   345→          const clone = el.cloneNode(true);
   346→          clone.removeAttribute('data-fid');
   347→          el.parentElement.insertBefore(clone, el.nextElementSibling);
   348→          assignIds();
   349→        }
   350→        break;
   351→      }
   352→      case 'add-component': {
   353→        const parent = document.querySelector('[data-fid="' + msg.data.parentId + '"]');
   354→        if (parent) {
   355→          const temp = document.createElement('div');
   356→          temp.innerHTML = msg.data.html;
   357→          const newEl = temp.firstElementChild;
   358→          if (msg.data.position >= 0 && parent.children[msg.data.position]) {
   359→            parent.insertBefore(newEl, parent.children[msg.data.position]);
   360→          } else {
   361→            parent.appendChild(newEl);
   362→          }
   363→          assignIds();
   364→        } else {
   365→          const temp = document.createElement('div');
   366→          temp.innerHTML = msg.data.html;
   367→          document.body.appendChild(temp.firstElementChild);
   368→          assignIds();
   369→        }
   370→        break;
   371→      }
   372→      case 'remove-element': {
   373→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   374→        if (el) {
   375→          el.remove();
   376→          if (selectedId === msg.data.elementId) hideSelection();
   377→          assignIds();
   378→        }
   379→        break;
   380→      }
   381→      case 'select-element': {
   382→        const el = document.querySelector('[data-fid="' + msg.data.elementId + '"]');
   383→        if (el) { showSelection(msg.data.elementId); sendElementInfo(el); }
   384→        break;
   385→      }
   386→      case 'highlight-element': { showHover(msg.data.elementId); break; }
   387→      case 'update-html': {
   388→        document.body.innerHTML = msg.data.html;
   389→        assignIds();
   390→        createOverlays();
   391→        hideSelection();
   392→        hideHover();
   393→        break;
   394→      }
   395→      case 'toggle-grid': {
   396→        if (gridEl) gridEl.style.display = msg.data.show ? 'block' : 'none';
   397→        break;
   398→      }
   399→      case 'apply-theme': {
   400→        const root = document.documentElement;
   401→        if (msg.data.colors) {
   402→          Object.entries(msg.data.colors).forEach(([key, val]) => {
   403→            root.style.setProperty('--' + key, val, 'important');
   404→            // Also update all elements with matching colors
   405→          });
   406→        }
   407→        if (msg.data.fontFamily) {
   408→          root.style.setProperty('--font-sans', msg.data.fontFamily, 'important');
   409→          document.querySelectorAll('*').forEach(el => {
   410→            const computed = getComputedStyle(el);
   411→            if (computed.fontFamily.includes('Inter') || computed.fontFamily.includes('system-ui') || computed.fontFamily.includes('sans-serif')) {
   412→              el.style.setProperty('font-family', msg.data.fontFamily, 'important');
   413→            }
   414→          });
   415→        }
   416→        break;
   417→      }
   418→      case 'get-elements-tree': {
   419→        function buildTree(el, depth) {
   420→          const fid = el.getAttribute('data-fid');
   421→          if (!fid) return null;
   422→          const children = [];
   423→          Array.from(el.children).forEach(child => {
   424→            const c = buildTree(child, depth + 1);
   425→            if (c) children.push(c);
   426→          });
   427→          return {
   428→            id: fid,
   429→            tag: el.tagName.toLowerCase(),
   430→            className: el.className || '',
   431→            textContent: (el.textContent || '').substring(0, 50),
   432→            children: children,
   433→            depth: depth
   434→          };
   435→        }
   436→        const tree = buildTree(document.body, 0);
   437→        window.parent.postMessage({ type: 'elements-tree', data: tree }, '*');
   438→        break;
   439→      }
   440→    }
   441→  });
   442→
   443→  assignIds();
   444→  createOverlays();
   445→  window.parent.postMessage({ type: 'ready', data: null }, '*');
   446→})();
   447→`
