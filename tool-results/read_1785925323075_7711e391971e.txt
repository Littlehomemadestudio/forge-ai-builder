    32→    32→/** Find a DOM node by its escaped path fid inside the canvas root. */
    33→    33→function findByFid(root: HTMLElement | null, fid: string): HTMLElement | null {
    34→    34→  if (!root) return null
    35→    35→  const parts = fid.split('-').map(Number)
    36→    36→  let el: Element = root
    37→    37→  for (const idx of parts) {
    38→    38→    const children = Array.from(el.children).filter((c) => c.nodeType === 1)
    39→    39→    if (idx >= children.length) return null
    40→    40→    el = children[idx]
    41→    41→  }
    42→    42→  return el as HTMLElement
    43→    43→}

   270→   270→              }

   276→   276→        }

   395→   395→    if (action === 'start-blank') { setHtml(STARTER); setCss(''); setHistory([STARTER]); setHistoryIndex(0); announce('Blank accessible page created') }
   396→   396→    else if (action === 'templates') navigate('builder')
   397→   397→    else if (action === 'ai') { announce('Describe your site in the AI bar below'); setAiFeedback('Tell the AI what to build, then press Go.') }
   398→   398→    else if (action === 'import') { announce('Import URL is not wired to a backend yet') }
   399→   399→    else if (action === 'paste') {
   400→   400→      const pasted = window.prompt('Paste your HTML below:')
   401→   401→      if (pasted && pasted.trim()) { setHtml(pasted.trim()); setCss(''); setHistory([pasted.trim()]); setHistoryIndex(0); announce('HTML imported') }
   402→   402→    }
   403→   403→  }, [navigate])
   404→   404→
   405→   405→  const onPublish = useCallback(() => {
   406→   406→    showToast('Publishing is not configured on this host yet.')
   407→   407→    announce('Publishing not configured')
   408→   408→  }, [showToast])
   409→   409→
   410→   410→  const onExport = useCallback(() => {
   411→   411→    const full = `<!doctype html><html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width, initial-scale=1"><title>${projectName}</title><style>${css}</style></head><body>${html}</body></html>`
   412→   412→    const blob = new Blob([full], { type: 'text/html' })
   413→   413→    const url = URL.createObjectURL(blob)
   414→   414→    const a = document.createElement('a'); a.href = url; a.download = 'index.html'; a.click()
   415→   415→    URL.revokeObjectURL(url)
   416→   416→    showToast('Exported index.html')
   417→   417→    announce('Exported index.html')
   418→   418→  }, [html, css, projectName, showToast])
   419→   419→
   420→   420→  const runCommand = useCallback((id: string) => {
   421→   421→    switch (id) {
   422→   422→      case 'undo': undo(); break
   423→   423→      case 'redo': redo(); break
   424→   424→      case 'duplicate': duplicate(); break
   425→   425→      case 'delete': remove(); break
   426→   426→      case 'add-section': commit(htmlRef.current + '<section style="padding:64px 24px"></section>', 'Add section'); break
   427→   427→      case 'add-text': commit(htmlRef.current + '<h2 style="padding:8px 0;color:#111827">New heading</h2>', 'Add text'); break
   428→   428→      case 'add-image': commit(htmlRef.current + '<img src="https://placehold.co/800x400?text=Image" alt="Placeholder image" style="max-width:100%">', 'Add image'); break
   429→   429→      case 'ai-generate': case 'ai-redesign': setCmdOpen(false); runAI(id === 'ai-generate' ? 'Generate a new section' : 'Redesign the page'); break
   430→   430→      case 'accessibility': runAudit(); break
   431→   431→      case 'publish': onPublish(); break
   432→   432→      case 'shortcuts': setShortcutsOpen(true); break
   433→   433→      case 'toggle-dark': setDarkMode((d) => !d); break
   434→   434→      case 'toggle-inspector': setInspectorOpen((o) => !o); break
   435→   435→    }
   436→   436→  }, [undo, redo, duplicate, remove, commit, runAI, runAudit, onPublish])
   437→   437→
   438→   438→  // ── Render ────────────────────────────────────────────────────────────
   439→   439→  const hasContent = html.trim().length > 0
   440→   440→  const baseFont = fontSizeScale === 'small' ? 14 : fontSizeScale === 'large' ? 18 : fontSizeScale === 'extra-large' ? 20 : 16
   441→   441→
   442→   442→  // Context-aware AI suggestions
   443→   443→  const aiSuggestions = React.useMemo(() => {
   444→   444→    const base = [{ id: 'improve', label: '✨ Improve selected' }]
   445→   445→    if (selection?.isImage) base.push({ id: 'alt', label: '✨ Add alt text' })
   446→   446→    if (selection?.isText) base.push({ id: 'rewrite', label: '✨ Rewrite text' })
   447→   447→    base.push({ id: 'contrast', label: '✨ Fix contrast' })
   448→   448→    return base
   449→   449→  }, [selection])
   450→   450→
   451→   451→  return (
   452→   452→    <div className={`ve-shell${darkMode ? ' ve-dark' : ''}`} style={{ height: '100vh', width: '100vw', display: 'flex', flexDirection: 'column', overflow: 'hidden', fontSize: baseFont }}>
   453→   453→      <LiveRegion />
   454→   454→
   455→   455→      {/* Saving indicator */}
   456→   456→      {saving && (
   457→   457→        <div aria-live="polite" style={{ position: 'fixed', top: 60, right: 16, zIndex: 600, background: COLORS.primary, color: '#FFF', padding: '6px 14px', borderRadius: RADIUS.lg, fontSize: 12, fontWeight: 500, display: 'flex', alignItems: 'center', gap: 6 }}>
   458→   458→          <span className="ve-spinner" style={{ width: 12, height: 12, border: '2px solid rgba(255,255,255,0.3)', borderTopColor: '#FFF', borderRadius: '50%', display: 'inline-block' }} />
   459→   459→          Saving…
   460→   460→        </div>
   461→   461→      )}
   462→   462→
   463→   463→      <TopNav
   464→   464→        projectName={projectName}
   465→   465→        onProjectNameChange={setProjectName}
   466→   466→        canUndo={historyIndex > 0}
   467→   467→        canRedo={historyIndex < history.length - 1}
   468→   468→        onUndo={undo}
   469→   469→        onRedo={redo}
   470→   470→        onBack={() => navigate('dashboard')}
   471→   471→        device={device}
   472→   472→        onDeviceChange={setDevice}
   473→   473→        zoom={zoom}
   474→   474→        onZoomIn={() => setZoom((z) => Math.min(200, z + 10))}
   475→   475→        onZoomOut={() => setZoom((z) => Math.max(20, z - 10))}
   476→   476→        onFit={() => setZoom(70)}
   477→   477→        onCommandPalette={() => setCmdOpen(true)}
   478→   478→        onShortcuts={() => setShortcutsOpen(true)}
   479→   479→        onPublish={onPublish}
   480→   480→        onToggleToolbar={() => setToolbarVisible((v) => !v)}
   481→   481→      />
   482→   482→
   483→   483→      <div style={{ flex: 1, display: 'flex', minHeight: 0 }}>
   484→   484→        {toolbarVisible && (
   485→   485→          <IconToolbar
   486→   486→            active={activeTool}
   487→   487→            onSelect={(id) => {
   488→   488→              setActiveTool(id)
   489→   489→              if (id === 'ai') announce('AI: describe a request in the assistant bar below')
   490→   490→            }}
   491→   491→          />
   492→   492→        )}
   493→   493→
   494→   494→        <div style={{ flex: 1, display: 'flex', flexDirection: 'column', minWidth: 0 }}>
   495→   495→          <div style={{ flex: 1, display: 'flex', position: 'relative', minHeight: 0 }}>
   496→   496→            {hasContent ? (
   497→   497→              <Canvas
   498→   498→                html={html}
   499→   499→                css={css}
   500→   500→                device={device}
   501→   501→                zoom={zoom}
   502→   502→                selectedFid={selection?.fid || null}
   503→   503→                spaceHeld={spaceHeld}
   504→   504→                onSelect={setSelection}
   505→   505→                onChangeHtml={commit}
   506→   506→                hasContent={hasContent}
   507→   507→                onContentReady={(el) => { contentRootRef.current = el }}
   508→   508→              />
   509→   509→            ) : (
   510→   510→              <div style={{ flex: 1, background: darkMode ? DARK_COLORS.background : COLORS.background, overflow: 'auto' }}>
   511→   511→                <EmptyCanvas onAction={onEmptyAction} />
   512→   512→              </div>
   513→   513→            )}
   514→   514→
   515→   515→            <FloatingSelectionBar
   516→   516→              selection={selection}
   517→   517→              onDuplicate={duplicate}
   518→   518→              onDelete={remove}
   519→   519→              onAI={() => runAI('Improve this element')}
   520→   520→              onBold={bold}
   521→   521→              onAlign={align}
   522→   522→            />
   523→   523→          </div>
   524→   524→
   525→   525→          <AIAssistantBar
   526→   526→            onSubmit={(prompt) => runAI(prompt)}
   527→   527→            isBusy={aiBusy}
   528→   528→            onStop={stopAI}
   529→   529→            onRegenerate={() => runAI('Improve this element')}
   530→   530→            lastResponse={aiFeedback}
   531→   531→            suggestions={aiSuggestions}
   532→   532→          />
   533→   533→          {aiFeedback && (
   534→   534→            <div role="status" aria-live="polite" style={{ padding: '8px 16px', background: darkMode ? DARK_COLORS.primaryLight : COLORS.primaryLight, color: darkMode ? DARK_COLORS.primary : COLORS.primary, fontSize: 13, borderTop: `1px solid ${darkMode ? DARK_COLORS.selectionLight : COLORS.selectionLight}` }}>
   535→   535→              ✨ {aiFeedback}
   536→   536→            </div>
   537→   537→          )}
   538→   538→          {auditResults.length > 0 && (
   539→   539→            <div role="status" style={{ padding: '8px 16px', background: darkMode ? DARK_COLORS.warningLight : COLORS.warningLight, color: '#92400E', fontSize: 13, borderTop: `1px solid ${darkMode ? DARK_COLORS.warning : COLORS.warning}` }}>
   540→   540→              <strong style={{ fontWeight: 600 }}>A11y audit:</strong> {auditResults.map((r) => <span key={r} style={{ display: 'inline-block', marginRight: 12 }}>• {r}</span>)}
   541→   541→            </div>
   542→   542→          )}
   543→   543→        </div>
   544→   544→
   545→   545→        {inspectorOpen && (
   546→   546→          <Inspector
   547→   547→            selection={selection}
   548→   548→            onApply={commitNode}
   549→   549→            onSelectText={onSelectText}
   550→   550→            onDeleteSelection={remove}
   551→   551→            onDuplicateSelection={duplicate}
   552→   552→            onRunAccessibilityAudit={runAudit}
   553→   553→            onFixAccessibility={fixAccessibility}
   554→   554→            onExport={onExport}
   555→   555→            onAI={(prompt) => runAI(prompt)}
   556→   556→            darkMode={darkMode}
   557→   557→            onToggleDarkMode={() => setDarkMode((d) => !d)}
   558→   558→            onToggleInspector={() => setInspectorOpen(false)}
   559→   559→          />
   560→   560→        )}
   561→   561→      </div>
   562→   562→
   563→   563→      <CommandPalette open={cmdOpen} onClose={() => setCmdOpen(false)} onRun={runCommand} />
   564→   564→      <ShortcutsHelp open={shortcutsOpen} onClose={() => setShortcutsOpen(false)} />
   565→   565→
   566→   566→      {toast && (
   567→   567→        <div role="status" style={{ position: 'fixed', bottom: 80, left: '50%', transform: 'translateX(-50%)', background: darkMode ? DARK_COLORS.text : COLORS.text, color: '#FFF', padding: '10px 18px', borderRadius: RADIUS.lg, boxShadow: SHADOWS.lg, zIndex: 500, fontSize: 14, fontWeight: 500 }}>
   568→   568→          {toast}
   569→   569→        </div>
   570→   570→      )}
   571→   571→    </div>
   572→   572→  )
   573→   573→}
