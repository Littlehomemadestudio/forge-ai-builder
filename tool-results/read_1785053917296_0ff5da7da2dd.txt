   215→function statusVariant(status: string) {
   216→  switch (status) {
   217→    case 'published': case 'live': case 'active': return 'default'
   218→    case 'draft': case 'building': case 'pending': return 'secondary'
   219→    case 'failed': return 'destructive'
   220→    default: return 'outline'
   221→  }
   222→}

   224→function statusIcon(status: string) {

   226→    case 'published': case 'live': case 'active': return <CheckCircle2 size={12} className="text-emerald-400" />
   227→    case 'draft': return <AlertCircle size={12} className="text-amber-400" />
